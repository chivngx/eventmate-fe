     1→"use client"
     2→
     3→import { useEffect, useState, useRef } from "react"
     4→import { useParams, useNavigate } from "@/lib/router"
     5→import { supabase } from "@/lib/supabase"
     6→import { getUserFacingMessage } from "@/lib/error"
     7→import { useUser } from "@/components/providers/AuthProvider"
     8→import MainLayout from "@/components/layout/MainLayout"
     9→import OrgLayout from "@/components/layout/OrgLayout"
    10→import { Send, MessageSquare, User, ArrowLeft, Calendar, Video, Clock, Check, X } from "lucide-react"
    11→import { Avatar, AvatarFallback, AvatarImage } from "@/components/ui/avatar"
    12→import { Button } from "@/components/ui/button"
    13→import InterviewModal from "@/components/chat/InterviewModal"
    14→
    15→interface ChatItem {
    16→  id: string
    17→  event_id: string
    18→  student_id: string
    19→  organizer_id: string
    20→  created_at: string
    21→  events: { title: string }
    22→  student_profile: { id: string; full_name: string; avatar_url: string }
    23→  organizer_profile: { id: string; full_name: string; avatar_url: string }
    24→}
    25→
    26→interface Message {
    27→  id: string
    28→  chat_id: string
    29→  sender_id: string
    30→  content: string
    31→  created_at: string
    32→}
    33→
    34→export default function Chat() {
    35→  const { id: routeChatId } = useParams<{ id: string }>()
    36→  const navigate = useNavigate()
    37→  const { user, role, isPremium, loading: authLoading } = useUser()
    38→  const currentUser = user
    39→
    40→  const [loading, setLoading] = useState(true)
    41→
    42→  const [chats, setChats] = useState<ChatItem[]>([])
    43→  const [activeChat, setActiveChat] = useState<ChatItem | null>(null)
    44→  const [messages, setMessages] = useState<Message[]>([])
    45→  const [newMessage, setNewMessage] = useState("")
    46→  const [sending, setSending] = useState(false)
    47→
    48→  const [messagesLimit, setMessagesLimit] = useState(20)
    49→  const [hasMoreMessages, setHasMoreMessages] = useState(true)
    50→
    51→  const [isInterviewModalOpen, setIsInterviewModalOpen] = useState(false)
    52→  const [interviewTitle, setInterviewTitle] = useState("")
    53→  const [interviewDate, setInterviewDate] = useState("")
    54→  const [interviewLink, setInterviewLink] = useState("")
    55→  const [interviews, setInterviews] = useState<Record<string, any>>({})
    56→  const [creatingInterview, setCreatingInterview] = useState(false)
    57→
    58→  const messagesEndRef = useRef<HTMLDivElement>(null)
    59→
    60→  useEffect(() => {
    61→    if (authLoading) return
    62→    if (!user) {
    63→      navigate("/login")
    64→      return
    65→    }
    66→    setLoading(false)
    67→  }, [user, authLoading, navigate])
    68→
    69→  const fetchChats = async () => {
    70→    if (!currentUser) return
    71→
    72→    const isOrg = role === "organizer"
    73→    const queryField = isOrg ? "organizer_id" : "student_id"
    74→
    75→    const { data, error } = await supabase
    76→      .from("chats")
    77→      .select(`
    78→        id, event_id, student_id, organizer_id, created_at,
    79→        events (title),
    80→        student_profile:profiles!student_id (id, full_name, avatar_url),
    81→        organizer_profile:profiles!organizer_id (id, full_name, avatar_url)
    82→      `)
    83→      .eq(queryField, currentUser?.id)
    84→      .order("created_at", { ascending: false })
    85→
    86→    if (error) {
    87→      console.error("Lỗi fetchChats:", error)
    88→    }
    89→
    90→    if (!error && data) {
    91→      const formattedChats = data.map((item: any) => ({
    92→        ...item,
    93→        events: Array.isArray(item.events) ? item.events[0] || { title: "Sự kiện chung" } : item.events || { title: "Sự kiện chung" },
    94→        student_profile: Array.isArray(item.student_profile) ? item.student_profile[0] || { id: "", full_name: "Người tham gia", avatar_url: "" } : item.student_profile || { id: "", full_name: "Người tham gia", avatar_url: "" },
    95→        organizer_profile: Array.isArray(item.organizer_profile) ? item.organizer_profile[0] || { id: "", full_name: "Ban tổ chức", avatar_url: "" } : item.organizer_profile || { id: "", full_name: "Ban tổ chức", avatar_url: "" }
    96→      })) as ChatItem[]
    97→
    98→      setChats(formattedChats)
    99→
   100→      if (routeChatId) {
   101→        const found = formattedChats.find(c => c.id === routeChatId)
   102→        if (found) {
   103→          setActiveChat(found)
   104→        } else {
   105→          const { data: directChat } = await supabase
   106→            .from("chats")
   107→            .select(`
   108→              id, event_id, student_id, organizer_id, created_at,
   109→              events (title),
   110→              student_profile:profiles!student_id (id, full_name, avatar_url),
   111→              organizer_profile:profiles!organizer_id (id, full_name, avatar_url)
   112→            `)
   113→            .eq("id", routeChatId)
   114→            .maybeSingle()
   115→
   116→          if (directChat) {
   117→            const formatted = {
   118→              ...directChat,
   119→              events: Array.isArray(directChat.events) ? directChat.events[0] || { title: "Sự kiện chung" } : directChat.events || { title: "Sự kiện chung" },
   120→              student_profile: Array.isArray(directChat.student_profile) ? directChat.student_profile[0] || { id: "", full_name: "Người tham gia", avatar_url: "" } : directChat.student_profile || { id: "", full_name: "Người tham gia", avatar_url: "" },
   121→              organizer_profile: Array.isArray(directChat.organizer_profile) ? directChat.organizer_profile[0] || { id: "", full_name: "Ban tổ chức", avatar_url: "" } : directChat.organizer_profile || { id: "", full_name: "Ban tổ chức", avatar_url: "" }
   122→            } as ChatItem
   123→            setActiveChat(formatted)
   124→          }
   125→        }
   126→      } else if (formattedChats.length > 0 && !activeChat) {
   127→        setActiveChat(formattedChats[0])
   128→      }
   129→    }
   130→  }
   131→
   132→  useEffect(() => {
   133→    if (currentUser) {
   134→      fetchChats()
   135→    }
   136→  }, [currentUser, role, routeChatId])
   137→
   138→  const fetchInterviews = async (chat: ChatItem) => {
   139→    const { data, error } = await supabase
   140→      .from("interviews")
   141→      .select("*")
   142→      .eq("student_id", chat.student_id)
   143→      .eq("organizer_id", chat.organizer_id)
   144→
   145→    if (!error && data) {
   146→      const map: Record<string, any> = {}
   147→      data.forEach((item) => {
   148→        map[item.id] = item
   149→      })
   150→      setInterviews(map)
   151→    }
   152→  }
   153→
   154→  const fetchMessages = async (chatId: string, currentLimit = 20) => {
   155→    const { data, error } = await supabase
   156→      .from("messages")
   157→      .select("*")
   158→      .eq("chat_id", chatId)
   159→      .order("created_at", { ascending: false })
   160→      .limit(currentLimit)
   161→
   162→    if (!error && data) {
   163→      const reversed = [...data].reverse()
   164→      setMessages(reversed)
   165→      setHasMoreMessages(data.length === currentLimit)
   166→    }
   167→  }
   168→
   169→  useEffect(() => {
   170→    if (activeChat) {
   171→      setMessagesLimit(20)
   172→      fetchMessages(activeChat.id, 20)
   173→      fetchInterviews(activeChat)
   174→
   175→      const channel = supabase
   176→        .channel(`chat-room-${activeChat.id}`)
   177→        .on("postgres_changes",
   178→          {
   179→            event: "INSERT",
   180→            schema: "public",
   181→            table: "messages"
   182→          },
   183→          (payload) => {
   184→            const newMsg = payload.new as Message
   185→            if (newMsg.chat_id === activeChat.id) {
   186→              setMessages((prev) => {
   187→                if (prev.some((m) => m.id === newMsg.id)) return prev
   188→                return [...prev, newMsg]
   189→              })
   190→
   191→              if (newMsg.content.startsWith("__INTERVIEW_REQUEST__:")) {
   192→                fetchInterviews(activeChat)
   193→              }
   194→
   195→              scrollToBottom()
   196→            }
   197→          }
   198→        )
   199→        .subscribe()
   200→
   201→      return () => {
   202→        supabase.removeChannel(channel)
   203→      }
   204→    }
   205→  }, [activeChat])
   206→
   207→  const scrollToBottom = () => {
   208→    setTimeout(() => {
   209→      messagesEndRef.current?.scrollIntoView({ behavior: "smooth" })
   210→    }, 100)
   211→  }
   212→
   213→  const handleSendMessage = async (e: React.FormEvent) => {
   214→    e.preventDefault()
   215→    if (!newMessage.trim() || !activeChat || !currentUser || sending) return
   216→
   217→    setSending(true)
   218→    const tempMsg = newMessage
   219→    setNewMessage("")
   220→
   221→    const { data, error } = await supabase
   222→      .from("messages")
   223→      .insert([
   224→        {
   225→          chat_id: activeChat.id,
   226→          sender_id: currentUser?.id,
   227→          content: tempMsg
   228→        }
   229→      ])
   230→      .select()
   231→      .single()
   232→
   233→    setSending(false)
   234→
   235→    if (error) {
   236→      console.error("Lỗi gửi tin nhắn:", error)
   237→      setNewMessage(tempMsg)
   238→    } else if (data) {
   239→      setMessages((prev) => {
   240→        if (prev.some((m) => m.id === data.id)) return prev
   241→        return [...prev, data as Message]
   242→      })
   243→      scrollToBottom()
   244→    }
   245→  }
   246→
   247→  const handleCreateInterview = async (e: React.FormEvent) => {
   248→    e.preventDefault()
   249→    if (!activeChat || !currentUser || !interviewTitle || !interviewDate || creatingInterview) return
   250→
   251→    setCreatingInterview(true)
   252→    try {
   253→      const { data: interview, error: intError } = await supabase
   254→        .from("interviews")
   255→        .insert([
   256→          {
   257→            event_id: activeChat.event_id,
   258→            organizer_id: currentUser?.id,
   259→            student_id: activeChat.student_id,
   260→            title: interviewTitle,
   261→            scheduled_at: new Date(interviewDate).toISOString(),
   262→            meeting_link: interviewLink || null,
   263→            status: "pending"
   264→          }
   265→        ])
   266→        .select()
   267→        .single()
   268→
   269→      if (intError) throw intError
   270→
   271→      const { data: messageData, error: msgError } = await supabase
   272→        .from("messages")
   273→        .insert([
   274→          {
   275→            chat_id: activeChat.id,
   276→            sender_id: currentUser?.id,
   277→            content: `__INTERVIEW_REQUEST__:${interview.id}`
   278→          }
   279→        ])
   280→        .select()
   281→        .single()
   282→
   283→      if (msgError) throw msgError
   284→
   285→      setInterviews(prev => ({ ...prev, [interview.id]: interview }))
   286→      if (messageData) {
   287→        setMessages(prev => [...prev, messageData as Message])
   288→      }
   289→
   290→      setInterviewTitle("")
   291→      setInterviewDate("")
   292→      setInterviewLink("")
   293→      setIsInterviewModalOpen(false)
   294→      scrollToBottom()
   295→    } catch (err: any) {
   296→      console.error("Lỗi tạo lịch hẹn:", err)
   297→      alert(getUserFacingMessage(err, "Không thể tạo lịch hẹn. Vui lòng thử lại."))
   298→    } finally {
   299→      setCreatingInterview(false)
   300→    }
   301→  }
   302→
   303→  const handleUpdateInterviewStatus = async (interviewId: string, newStatus: "accepted" | "rejected") => {
   304→    try {
   305→      const interview = interviews[interviewId]
   306→      if (!interview) return
   307→
   308→      const { error } = await supabase
   309→        .from("interviews")
   310→        .update({ status: newStatus })
   311→        .eq("id", interviewId)
   312→
   313→      if (error) throw error
   314→
   315→      setInterviews(prev => ({
   316→        ...prev,
   317→        [interviewId]: { ...prev[interviewId], status: newStatus }
   318→      }))
   319→
   320→      await supabase.from("notifications").insert([
   321→        {
   322→          user_id: interview.organizer_id,
   323→          title: newStatus === "accepted" ? "Lịch phỏng vấn được chấp nhận" : "Lịch phỏng vấn bị từ chối",
   324→          message: `Người tham gia đã ${newStatus === "accepted" ? "chấp nhận" : "từ chối"} lịch hẹn phỏng vấn: ${interview.title}`,
   325→          is_read: false
   326→        }
   327→      ])
   328→    } catch (err: any) {
   329→      console.error("Lỗi cập nhật lịch hẹn:", err)
   330→      alert(getUserFacingMessage(err, "Không thể cập nhật trạng thái. Vui lòng thử lại."))
   331→    }
   332→  }
   333→
   334→  if (loading) {
   335→    return (
   336→      <div className="flex h-screen w-screen items-center justify-center bg-background">
   337→        <div className="h-8 w-8 animate-spin rounded-full border-4 border-primary border-t-transparent"></div>
   338→      </div>
   339→    )
   340→  }
   341→
   342→  const getPartnerProfile = (chat: ChatItem) => {
   343→    return role === "organizer" ? chat.student_profile : chat.organizer_profile
   344→  }
   345→
   346→  const chatUI = (
   347→    <div className="mx-auto flex h-[calc(100vh-9rem)] w-full max-w-6xl overflow-hidden rounded-xl border border-border bg-card shadow-sm lg:h-[calc(100vh-7rem)]">
   348→      {/* LEFT: chat list */}
   349→      <section
   350→        className={`w-full flex-col border-r border-border md:flex md:w-80 ${activeChat ? "hidden md:flex" : "flex"}`}
   351→        aria-label="Danh sách hội thoại"
   352→      >
   353→        <header className="border-b border-border p-4">
   354→          <h1 className="flex items-center gap-2 text-base font-semibold text-foreground">
   355→            <MessageSquare className="h-4 w-4 text-slate-600" />
   356→            Hội thoại
   357→          </h1>
   358→          <p className="mt-0.5 text-xs text-muted-foreground">{chats.length} cuộc trò chuyện</p>
   359→        </header>
   360→        <div className="flex-1 space-y-1 overflow-y-auto p-2">
   361→          {chats.length === 0 ? (
   362→            <div className="px-3 py-12 text-center text-xs text-muted-foreground">
   363→              Chưa có hội thoại nào
   364→            </div>
   365→          ) : (
   366→            chats.map((chat) => {
   367→              const partner = getPartnerProfile(chat)
   368→              const isActive = activeChat?.id === chat.id
   369→              return (
   370→                <button
   371→                  key={chat.id}
   372→                  onClick={() => {
   373→                    setActiveChat(chat)
   374→                    navigate(`/chat/${chat.id}`)
   375→                  }}
   376→                  className={`flex w-full items-center gap-3 rounded-lg p-3 text-left transition-colors ${isActive
   377→                    ? "bg-slate-100"
   378→                    : "hover:bg-muted border border-transparent"
   379→                    }`}
   380→                >
   381→                  <Avatar className="h-10 w-10 shrink-0 border border-border">
   382→                    <AvatarImage src={partner.avatar_url} />
   383→                    <AvatarFallback className="bg-muted text-sm font-semibold text-foreground">
   384→                      {partner.full_name?.charAt(0).toUpperCase() || <User className="h-4 w-4" />}
   385→                    </AvatarFallback>
   386→                  </Avatar>
   387→                  <div className="min-w-0 flex-1">
   388→                    <h4 className={`truncate text-sm font-semibold ${isActive ? "text-slate-600" : "text-foreground"}`}>
   389→                      {partner.full_name}
   390→                    </h4>
   391→                    <p className="truncate text-xs text-muted-foreground">
   392→                      {chat.events.title}
   393→                    </p>
   394→                  </div>
   395→                </button>
   396→              )
   397→            })
   398→          )}
   399→        </div>
   400→      </section>
   401→
   402→      {/* RIGHT: message thread */}
   403→      <section
   404→        className={`flex-1 flex-col ${!activeChat ? "hidden md:flex md:items-center md:justify-center" : "flex"}`}
   405→        aria-label="Luồng tin nhắn"
   406→      >
   407→        {activeChat ? (
   408→          <>
   409→            {/* Thread header */}
   410→            <header className="flex items-center justify-between border-b border-border bg-card p-3 sm:p-4">
   411→              <div className="flex min-w-0 items-center gap-3">
   412→                <button
   413→                  onClick={() => {
   414→                    setActiveChat(null)
   415→                    navigate("/chat")
   416→                  }}
   417→                  aria-label="Quay lại danh sách hội thoại"
   418→                  className="mr-1 shrink-0 rounded-lg p-1.5 text-muted-foreground transition-colors hover:bg-slate-100 hover:text-foreground md:hidden"
   419→                >
   420→                  <ArrowLeft className="h-5 w-5" />
   421→                </button>
   422→                <Avatar className="h-9 w-9 shrink-0 border border-border">
   423→                  <AvatarImage src={getPartnerProfile(activeChat).avatar_url} />
   424→                  <AvatarFallback className="bg-muted text-sm font-semibold text-foreground">
   425→                    {getPartnerProfile(activeChat).full_name?.charAt(0).toUpperCase() || <User className="h-4 w-4" />}
   426→                  </AvatarFallback>
   427→                </Avatar>
   428→                <div className="min-w-0">
   429→                  <h2 className="truncate text-sm font-semibold text-foreground">
   430→                    {getPartnerProfile(activeChat).full_name}
   431→                  </h2>
   432→                  <p className="truncate text-xs text-muted-foreground">
   433→                    {activeChat.events.title}
   434→                  </p>
   435→                </div>
   436→              </div>
   437→
   438→              {role === "organizer" && (
   439→                <Button
   440→                  onClick={() => setIsInterviewModalOpen(true)}
   441→                  className="shrink-0 rounded-lg bg-primary px-3 text-xs font-medium text-primary-foreground hover:bg-primary/90"
   442→                >
   443→                  <Calendar className="h-3.5 w-3.5" />
   444→                  <span className="hidden sm:inline">Hẹn phỏng vấn</span>
   445→                </Button>
   446→              )}
   447→            </header>
   448→
   449→            {/* Messages */}
   450→            <div className="flex-1 space-y-3 overflow-y-auto bg-muted/30 p-3 sm:p-4">
   451→              {hasMoreMessages && (
   452→                <div className="flex justify-center pb-2">
   453→                  <button
   454→                    onClick={() => {
   455→                      const nextLimit = messagesLimit + 20
   456→                      setMessagesLimit(nextLimit)
   457→                      fetchMessages(activeChat.id, nextLimit)
   458→                    }}
   459→                    className="rounded-full bg-card px-3 py-1 text-xs font-medium text-slate-600 border border-border hover:bg-slate-100 transition-colors"
   460→                  >
   461→                    Tải tin nhắn cũ hơn
   462→                  </button>
   463→                </div>
   464→              )}
   465→
   466→              {messages.map((msg) => {
   467→                const isMe = msg.sender_id === currentUser?.id
   468→
   469→                // Interview request message
   470→                if (msg.content.startsWith("__INTERVIEW_REQUEST__:")) {
   471→                  const interviewId = msg.content.substring("__INTERVIEW_REQUEST__:".length)
   472→                  const interview = interviews[interviewId]
   473→
   474→                  if (!interview) {
   475→                    return (
   476→                      <div key={msg.id} className="flex justify-center py-2">
   477→                        <div className="rounded-lg bg-muted px-3 py-1.5 text-xs text-muted-foreground">
   478→                          Đang tải thông tin phỏng vấn...
   479→                        </div>
   480→                      </div>
   481→                    )
   482→                  }
   483→
   484→                  const isStudent = role === "student"
   485→
   486→                  return (
   487→                    <div key={msg.id} className="flex justify-center py-3">
   488→                      <div className="w-full max-w-md rounded-xl border border-border bg-card p-4 shadow-sm">
   489→                        <div className="flex items-center gap-3 border-b border-border pb-3">
   490→                          <div className="rounded-lg bg-slate-100 p-2 text-slate-600">
   491→                            <Calendar className="h-4 w-4" />
   492→                          </div>
   493→                          <div>
   494→                            <h4 className="text-sm font-semibold text-foreground">Lời mời phỏng vấn</h4>
   495→                            <p className="text-xs text-muted-foreground">Hẹn lịch trực tuyến</p>
   496→                          </div>
   497→                        </div>
   498→
   499→                        <div className="space-y-2 py-3 text-xs">
   500→                          <div>
   501→                            <span className="block text-muted-foreground">Chủ đề</span>
   502→                            <span className="font-medium text-foreground">{interview.title}</span>
   503→                          </div>
   504→                          <div>
   505→                            <span className="block text-muted-foreground">Thời gian</span>
   506→                            <span className="flex items-center gap-1.5 font-medium text-foreground">
   507→                              <Clock className="h-3.5 w-3.5 text-slate-600" />
   508→                              {new Date(interview.scheduled_at).toLocaleString("vi-VN", {
   509→                                dateStyle: "medium",
   510→                                timeStyle: "short"
   511→                              })}
   512→                            </span>
   513→                          </div>
   514→                          {interview.meeting_link && (
   515→                            <div>
   516→                              <span className="block text-muted-foreground">Link phòng họp</span>
   517→                              <a
   518→                                href={interview.meeting_link.startsWith("http") ? interview.meeting_link : `https://${interview.meeting_link}`}
   519→                                target="_blank"
   520→                                rel="noopener noreferrer"
   521→                                className="inline-flex items-center gap-1.5 font-medium text-slate-600 hover:underline"
   522→                              >
   523→                                <Video className="h-3.5 w-3.5" />
   524→                                Tham gia cuộc gọi
   525→                              </a>
   526→                            </div>
   527→                          )}
   528→                        </div>
   529→
   530→                        <div className="border-t border-border pt-3">
   531→                          {interview.status === "pending" ? (
   532→                            isStudent ? (
   533→                              <div className="flex gap-2">
   534→                                <Button
