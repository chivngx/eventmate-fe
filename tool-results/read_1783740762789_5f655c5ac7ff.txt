     1→"use client"
     2→
     3→import { useState, useEffect } from "react"
     4→import { useParams, useNavigate } from "@/lib/router"
     5→import { supabase } from "@/lib/supabase"
     6→import MainLayout from "@/components/layout/MainLayout"
     7→import { MapPin, Calendar, CheckCircle, XCircle, Clock3, Bookmark, Briefcase, Tag, DollarSign, Users, ExternalLink } from "lucide-react"
     8→import { Button } from "@/components/ui/button"

    11→

    20→    const [isApplying, setIsApplying] = useState(false)
    21→    const [isBookmarked, setIsBookmarked] = useState(false)
    22→

    33→            } else {

    35→            }

    41→

    43→                if (isUuid && eventData.slug) {

    52→
    53→                    if (profile?.role === "student") {

    55→                            .from("applications")

    57→                            .eq("event_id", eventData.id)
    58→                            .eq("student_id", user.id)
    59→                            .maybeSingle()

    63→                        const { data: bookmarkData } = await supabase
    64→                            .from("event_bookmarks")
    65→                            .select("id")
    66→                            .eq("event_id", eventData.id)
    67→                            .eq("student_id", user.id)

    70→                        if (bookmarkData) setIsBookmarked(true)
    71→                    }
    72→                }

    75→            }
    76→            setLoading(false)
    77→        }
    78→
    79→        fetchEventDetails()
    80→    }, [id])

    86→        if (!user) {
    87→            window.dispatchEvent(new CustomEvent("open-auth-modal", { detail: { mode: "login" } }))
    88→            return

    92→            { event_id: event.id, student_id: user.id }

    98→            setApplyStatus('pending')

   100→        }

   102→    }

   107→            window.dispatchEvent(new CustomEvent("open-auth-modal", { detail: { mode: "login" } }))
   108→            return
   109→        }

   113→                .from("event_bookmarks")
   114→                .delete()

   116→                .eq("event_id", event.id)
   117→
   118→            if (!error) {
   119→                setIsBookmarked(false)
   120→                showToast({ title: "Đã hủy lưu", message: "Đã hủy lưu tin tuyển dụng thành công.", type: "info" })

   123→        } else {

   125→                .from("event_bookmarks")
   126→                .insert([{ student_id: user.id, event_id: event.id }])
   127→
   128→            if (!error) {
   129→                setIsBookmarked(true)
   130→                showToast({ title: "Đã lưu tin", message: "Đã lưu tin tuyển dụng thành công.", type: "success" })

   143→                <Button onClick={() => navigate(-1)} className="mt-4 rounded-xl bg-slate-900 text-white">Quay lại</Button>

   145→        </MainLayout>
   146→    )
   147→
   148→    const isPastDeadline = event.application_deadline ? new Date() > new Date(event.application_deadline) : false;
   149→    const disabledApply = isApplying || event.status !== 'upcoming';

   157→                    <CheckCircle className="w-4 h-4 mr-2" /> Trúng tuyển
   158→                </button>

   163→                <button disabled className="job-detail_info--actions-button button-primary btn-apply-job flex items-center justify-center rounded-md font-semibold bg-rose-50 border border-rose-100 text-rose-500 h-[40px] px-6 min-[1440px]:w-[569px] min-[1440px]:h-[40px] min-[1440px]:rounded-[6px] min-[1440px]:text-[14px] min-[1440px]:gap-[6px] min-[1440px]:leading-[22px] min-[1440px]:p-[8px_16px_8px_12px]">

   181→            >
   182→                {isApplying ? "Đang xử lý..." : isPastDeadline ? "Đã hết hạn nộp đơn" : event.status !== 'upcoming' ? "Đã đóng đăng ký" : "Ứng tuyển ngay"}
   183→            </button>
   184→        )
   185→    }
   186→
   187→    return (
   188→        <MainLayout role={role}>
   189→            <div className="max-w-6xl mx-auto pt-1 pb-6 px-4 animate-in fade-in slide-in-from-bottom-4 duration-500">
   190→                {/* Two Column Layout */}
   191→                <div className="job-detail_body flex flex-col lg:flex-row gap-6 items-start min-[1440px]:w-[1140px] min-[1440px]:gap-[28px] min-[1440px]:flex-row">
   192→                    {/* Left Column (Width: approx 760px on large screen) */}
   193→                    <div className="job-detail_body-left flex-1 w-full lg:max-w-[760px] space-y-6 min-[1440px]:w-[761px] min-[1440px]:gap-[28px] min-[1440px]:flex-col">
   194→                        {/* Box 1: Header / General Summary */}
   195→                        <div id="header-job-info" className="job-detail_box job-detail_info bg-white rounded-lg border border-slate-200 p-6 shadow-sm min-[1440px]:p-[20px_24px] min-[1440px]:rounded-[8px] min-[1440px]:gap-[16px] min-[1440px]:flex-col">
   196→                            <h1 className="job-detail_info--title text-xl md:text-2xl font-bold text-[#263a4d] leading-tight mb-4 min-[1440px]:mb-[16px] min-[1440px]:w-[713px] min-[1440px]:text-[20px] min-[1440px]:font-bold min-[1440px]:font-sans min-[1440px]:tracking-[-0.2px] min-[1440px]:leading-[28px]">
   197→                                {event.title}
   198→                            </h1>
   199→
   200→                            {/* Quick Info Grid */}
   201→                            <div className="job-detail_info--sections grid grid-cols-1 md:grid-cols-3 gap-4 border-t border-slate-100 pt-5 pb-6 min-[1440px]:w-[713px] min-[1440px]:h-[46px] min-[1440px]:flex-row min-[1440px]:items-center min-[1440px]:justify-between min-[1440px]:pt-0 min-[1440px]:pb-0 min-[1440px]:border-t-0 min-[1440px]:flex">
   202→                                <div className="job-detail_info--section section-salary flex items-center gap-3 min-[1440px]:h-[46px] min-[1440px]:flex-row min-[1440px]:items-center min-[1440px]:gap-[16px]">
   203→                                    <div className="job-detail_info--section-icon w-9 h-9 rounded-full bg-[#00b14f]/5 flex items-center justify-center shrink-0 min-[1440px]:w-[40px] min-[1440px]:h-[40px] min-[1440px]:rounded-[30px] min-[1440px]:p-[10px] min-[1440px]:flex min-[1440px]:flex-col min-[1440px]:justify-center min-[1440px]:items-center min-[1440px]:gap-[10px] min-[1440px]:bg-gradient-to-br min-[1440px]:from-[#00bf5d] min-[1440px]:to-[#00907c]">
   204→                                        <DollarSign className="w-4 h-4 text-[#00b14f] min-[1440px]:text-white" />
   205→                                    </div>
   206→                                    <div>
   207→                                        <p className="job-detail_info--section-content-title text-[12px] text-slate-400 font-medium min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#263a4d] min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.14px]">Quyền lợi / Lương</p>
   208→                                        <p className="job-detail_info--section-content-value text-[13px] text-[#212f3f] font-semibold truncate max-w-[180px] min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#212f3f] min-[1440px]:font-semibold min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.175px]" title={event.benefits || "Thỏa thuận"}>
   209→                                            {event.benefits || "Thỏa thuận"}
   210→                                        </p>
   211→                                    </div>
   212→                                </div>
   213→                                <div className="job-detail_info--section section-location flex items-center gap-3 min-[1440px]:flex-row min-[1440px]:items-center min-[1440px]:gap-[16px] min-[1440px]:flex-basis-0">
   214→                                    <div className="job-detail_info--section-icon w-9 h-9 rounded-full bg-[#00b14f]/5 flex items-center justify-center shrink-0 min-[1440px]:w-[40px] min-[1440px]:h-[40px] min-[1440px]:rounded-[30px] min-[1440px]:p-[10px] min-[1440px]:flex min-[1440px]:flex-col min-[1440px]:justify-center min-[1440px]:items-center min-[1440px]:gap-[10px] min-[1440px]:bg-gradient-to-br min-[1440px]:from-[#00bf5d] min-[1440px]:to-[#00907c]">
   215→                                        <MapPin className="w-4 h-4 text-[#00b14f] min-[1440px]:text-white" />
   216→                                    </div>
   217→                                    <div>
   218→                                        <p className="job-detail_info--section-content-title text-[12px] text-slate-400 font-medium min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#263a4d] min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.14px]">Khu vực</p>
   219→                                        <p className="job-detail_info--section-content-value text-[13px] text-[#212f3f] font-semibold truncate min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#212f3f] min-[1440px]:font-semibold min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.175px]" title={event.danang_wards?.name ? `P. ${event.danang_wards.name}` : "Đà Nẵng"}>
   220→                                            {event.danang_wards?.name ? `${event.danang_wards.name}` : "Đà Nẵng"}
   221→                                        </p>
   222→                                    </div>
   223→                                </div>
   224→                                <div className="job-detail_info--section section-eventdate flex items-center gap-3 min-[1440px]:flex-row min-[1440px]:items-center min-[1440px]:gap-[16px]">
   225→                                    <div className="job-detail_info--section-icon w-9 h-9 rounded-full bg-[#00b14f]/5 flex items-center justify-center shrink-0 min-[1440px]:w-[40px] min-[1440px]:h-[40px] min-[1440px]:rounded-[30px] min-[1440px]:p-[10px] min-[1440px]:flex min-[1440px]:flex-col min-[1440px]:justify-center min-[1440px]:items-center min-[1440px]:gap-[10px] min-[1440px]:bg-gradient-to-br min-[1440px]:from-[#00bf5d] min-[1440px]:to-[#00907c]">
   226→                                        <Calendar className="w-4 h-4 text-[#00b14f] min-[1440px]:text-white" />
   227→                                    </div>
   228→                                    <div>
   229→                                        <p className="job-detail_info--section-content-title text-[12px] text-slate-400 font-medium min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#263a4d] min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.14px]">Ngày diễn ra</p>
   230→                                        <p className="job-detail_info--section-content-value text-[13px] text-[#212f3f] font-semibold min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#212f3f] min-[1440px]:font-semibold min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.175px]">
   231→                                            {event.event_date ? new Date(event.event_date).toLocaleDateString('vi-VN') : "Đang cập nhật"}

   237→                            {/* Hạn chót nộp hồ sơ */}
   238→                            <div className="job-detail_info--flex flex items-center text-sm text-[#333] gap-2 mt-4 mb-6 min-[1440px]:w-[713px] min-[1440px]:h-[26px] min-[1440px]:flex min-[1440px]:items-center min-[1440px]:gap-[4px] min-[1440px]:text-[14px] min-[1440px]:leading-[20px] min-[1440px]:text-[#333] min-[1440px]:mt-[16px] min-[1440px]:mb-6">
   239→                                <div className="job-detail_info--deadline flex items-center gap-1 text-sm text-[#7f878f] bg-slate-50 px-2.5 py-1.5 md:p-[2px_8px_2px_4px] rounded min-[1440px]:rounded-[4px] min-[1440px]:gap-[4px] min-[1440px]:text-[14px] min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.14px] min-[1440px]:text-[#7f878f]">

   246→                                    <span className="deadline text-sm font-semibold text-[#263a4d] min-[1440px]:text-[14px] min-[1440px]:font-semibold min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.175px] min-[1440px]:text-[#263a4d]">
