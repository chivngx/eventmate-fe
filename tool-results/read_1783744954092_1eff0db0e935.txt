     1→"use client"
     2→
     3→import { useState, useEffect } from "react"
     4→import { useParams, useNavigate } from "@/lib/router"
     5→import { supabase } from "@/lib/supabase"
     6→import { getUserFacingMessage } from "@/lib/error"
     7→import MainLayout from "@/components/layout/MainLayout"
     8→import { MapPin, Calendar, CheckCircle, XCircle, Clock3, Bookmark, Briefcase, Tag, DollarSign, Users, ExternalLink } from "lucide-react"
     9→import { Button } from "@/components/ui/button"
    10→import { useToast } from "@/components/ui/ToastProvider"
    11→import { SkeletonEventDetail } from "@/components/ui/Skeleton"
    12→
    13→export default function EventDetail() {
    14→    const { id } = useParams<{ id: string }>()

    18→    const [loading, setLoading] = useState(true)
    19→    const [role, setRole] = useState<string>("guest")

    21→    const [isApplying, setIsApplying] = useState(false)
    22→    const [isBookmarked, setIsBookmarked] = useState(false)
    23→
    24→    useEffect(() => {

    26→            setLoading(true)

    29→                .from("events")
    30→                .select("*, profiles(id, full_name, avatar_url, slug, scale, address), danang_wards(name)")
    31→

    33→                eventQuery = eventQuery.eq("id", id)
    34→            } else {
    35→                eventQuery = eventQuery.eq("slug", id)
    36→            }
    37→
    38→            const { data: eventData, error } = await eventQuery.maybeSingle()
    39→
    40→            if (eventData) {
    41→                setEvent(eventData)
    42→
    43→                // Tự động chuyển hướng URL từ ID dạng UUID sang dạng Slug SEO thân thiện
    44→                if (isUuid && eventData.slug) {
    45→                    navigate(`/jobs/${eventData.slug}`, { replace: true })
    46→                }
    47→
    48→                const { data: { user } } = await supabase.auth.getUser()
    49→
    50→                if (user) {
    51→                    const { data: profile } = await supabase.from("profiles").select("role").eq("id", user.id).maybeSingle()
    52→                    if (profile) setRole(profile.role)
    53→
    54→                    if (profile?.role === "student") {
    55→                        const { data: appData } = await supabase
    56→                            .from("applications")
    57→                            .select("status")
    58→                            .eq("event_id", eventData.id)
    59→                            .eq("student_id", user.id)
    60→                            .maybeSingle()
    61→
    62→                        if (appData) setApplyStatus(appData.status)
    63→
    64→                        const { data: bookmarkData } = await supabase
    65→                            .from("event_bookmarks")
    66→                            .select("id")
    67→                            .eq("event_id", eventData.id)
    68→                            .eq("student_id", user.id)
    69→                            .maybeSingle()
    70→
    71→                        if (bookmarkData) setIsBookmarked(true)
    72→                    }
    73→                }
    74→            } else {
    75→                console.error("Lỗi hoặc không tìm thấy sự kiện", error)
    76→            }
    77→            setLoading(false)
    78→        }
    79→
    80→        fetchEventDetails()
    81→    }, [id])
    82→

    84→        setIsApplying(true)
    85→        const { data: { user } } = await supabase.auth.getUser()
    86→
    87→        if (!user) {
    88→            window.dispatchEvent(new CustomEvent("open-auth-modal", { detail: { mode: "login" } }))
    89→            return
    90→        }
    91→
    92→        const { error } = await supabase.from("applications").insert([
    93→            { event_id: event.id, student_id: user.id }
    94→        ])
    95→

    97→            showToast({ title: "Đã xảy ra lỗi", message: getUserFacingMessage(error, "Vui lòng thử lại."), type: "error" })

    99→            setApplyStatus('pending')
   100→            showToast({ title: "Ứng tuyển thành công", message: "Đơn ứng tuyển của bạn đã được gửi. Vui lòng chờ BTC phê duyệt.", type: "success" })
   101→        }
   102→        setIsApplying(false)
   103→    }

   106→        const { data: { user } } = await supabase.auth.getUser()

   108→            window.dispatchEvent(new CustomEvent("open-auth-modal", { detail: { mode: "login" } }))
   109→            return
   110→        }
   111→
   112→        if (isBookmarked) {
   113→            const { error } = await supabase
   114→                .from("event_bookmarks")
   115→                .delete()
   116→                .eq("student_id", user.id)
   117→                .eq("event_id", event.id)
   118→
   119→            if (!error) {
   120→                setIsBookmarked(false)
   121→                showToast({ title: "Đã hủy lưu", message: "Đã hủy lưu tin tuyển dụng thành công.", type: "info" })
   122→            }
   123→            else showToast({ title: "Đã xảy ra lỗi", message: getUserFacingMessage(error, "Vui lòng thử lại."), type: "error" })
   124→        } else {
   125→            const { error } = await supabase
   126→                .from("event_bookmarks")
   127→                .insert([{ student_id: user.id, event_id: event.id }])
   128→
   129→            if (!error) {
   130→                setIsBookmarked(true)
   131→                showToast({ title: "Đã lưu tin", message: "Đã lưu tin tuyển dụng thành công.", type: "success" })
   132→            }
   133→            else showToast({ title: "Đã xảy ra lỗi", message: getUserFacingMessage(error, "Vui lòng thử lại."), type: "error" })
   134→        }
   135→    }

   137→    if (loading) return <SkeletonEventDetail />

   141→            <div className="text-center py-20">
   142→                <h2 className="text-2xl font-black text-slate-900">Không tìm thấy sự kiện</h2>
   143→                <p className="text-slate-500 mt-2">Sự kiện này có thể đã bị xóa hoặc không tồn tại.</p>
   144→                <Button onClick={() => navigate(-1)} className="mt-4 rounded-xl bg-slate-900 text-white">Quay lại</Button>
   145→            </div>
   146→        </MainLayout>
   147→    )
   148→
   149→    const isPastDeadline = event.application_deadline ? new Date() > new Date(event.application_deadline) : false;
   150→    const disabledApply = isApplying || event.status !== 'upcoming';
   151→
   152→    const renderApplyAction = () => {
   153→        if (role !== "student") return null
   154→
   155→        if (applyStatus === 'approved') {
   156→            return (
   157→                <button disabled className="job-detail_info--actions-button button-primary btn-apply-job flex items-center justify-center rounded-md font-semibold bg-emerald-100 border border-emerald-200 text-emerald-700 h-[40px] px-6 min-[1440px]:w-[569px] min-[1440px]:h-[40px] min-[1440px]:rounded-[6px] min-[1440px]:text-[14px] min-[1440px]:gap-[6px] min-[1440px]:leading-[22px] min-[1440px]:p-[8px_16px_8px_12px]">
   158→                    <CheckCircle className="w-4 h-4 mr-2" /> Trúng tuyển
   159→                </button>
   160→            )
   161→        }
   162→        if (applyStatus === 'rejected') {
   163→            return (
   164→                <button disabled className="job-detail_info--actions-button button-primary btn-apply-job flex items-center justify-center rounded-md font-semibold bg-rose-50 border border-rose-100 text-rose-500 h-[40px] px-6 min-[1440px]:w-[569px] min-[1440px]:h-[40px] min-[1440px]:rounded-[6px] min-[1440px]:text-[14px] min-[1440px]:gap-[6px] min-[1440px]:leading-[22px] min-[1440px]:p-[8px_16px_8px_12px]">

   173→                </button>
   174→            )
   175→        }
   176→
   177→        return (
   178→            <button
   179→                onClick={handleApply}
   180→                disabled={disabledApply || isPastDeadline}
   181→                className="job-detail_info--actions-button button-primary open-apply-modal btn-apply-job flex items-center justify-center bg-[#00b14f] hover:bg-[#009a44] text-white font-semibold font-sans rounded-md transition-all active:scale-95 disabled:opacity-50 h-[40px] px-6 text-sm flex-1 cursor-pointer min-[1440px]:w-[569px] min-[1440px]:h-[40px] min-[1440px]:rounded-[6px] min-[1440px]:text-[14px] min-[1440px]:font-semibold min-[1440px]:gap-[6px] min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.175px] min-[1440px]:p-[8px_16px_8px_12px]"
   182→            >
   183→                {isApplying ? "Đang xử lý..." : isPastDeadline ? "Đã hết hạn nộp đơn" : event.status !== 'upcoming' ? "Đã đóng đăng ký" : "Ứng tuyển ngay"}
   184→            </button>
   185→        )
   186→    }
   187→
   188→    return (
   189→        <MainLayout role={role}>
   190→            <div className="max-w-6xl mx-auto pt-1 pb-6 px-4 animate-in fade-in slide-in-from-bottom-4 duration-500">
   191→                {/* Two Column Layout */}
   192→                <div className="job-detail_body flex flex-col lg:flex-row gap-6 items-start min-[1440px]:w-[1140px] min-[1440px]:gap-[28px] min-[1440px]:flex-row">
   193→                    {/* Left Column (Width: approx 760px on large screen) */}
   194→                    <div className="job-detail_body-left flex-1 w-full lg:max-w-[760px] space-y-6 min-[1440px]:w-[761px] min-[1440px]:gap-[28px] min-[1440px]:flex-col">
   195→                        {/* Box 1: Header / General Summary */}
   196→                        <div id="header-job-info" className="job-detail_box job-detail_info bg-white rounded-lg border border-slate-200 p-6 shadow-sm min-[1440px]:p-[20px_24px] min-[1440px]:rounded-[8px] min-[1440px]:gap-[16px] min-[1440px]:flex-col">
   197→                            <h1 className="job-detail_info--title text-xl md:text-2xl font-bold text-[#263a4d] leading-tight mb-4 min-[1440px]:mb-[16px] min-[1440px]:w-[713px] min-[1440px]:text-[20px] min-[1440px]:font-bold min-[1440px]:font-sans min-[1440px]:tracking-[-0.2px] min-[1440px]:leading-[28px]">
   198→                                {event.title}
   199→                            </h1>
   200→
   201→                            {/* Quick Info Grid */}
   202→                            <div className="job-detail_info--sections grid grid-cols-1 md:grid-cols-3 gap-4 border-t border-slate-100 pt-5 pb-6 min-[1440px]:w-[713px] min-[1440px]:h-[46px] min-[1440px]:flex-row min-[1440px]:items-center min-[1440px]:justify-between min-[1440px]:pt-0 min-[1440px]:pb-0 min-[1440px]:border-t-0 min-[1440px]:flex">
   203→                                <div className="job-detail_info--section section-salary flex items-center gap-3 min-[1440px]:h-[46px] min-[1440px]:flex-row min-[1440px]:items-center min-[1440px]:gap-[16px]">
   204→                                    <div className="job-detail_info--section-icon w-9 h-9 rounded-full bg-[#00b14f]/5 flex items-center justify-center shrink-0 min-[1440px]:w-[40px] min-[1440px]:h-[40px] min-[1440px]:rounded-[30px] min-[1440px]:p-[10px] min-[1440px]:flex min-[1440px]:flex-col min-[1440px]:justify-center min-[1440px]:items-center min-[1440px]:gap-[10px] min-[1440px]:bg-gradient-to-br min-[1440px]:from-[#00bf5d] min-[1440px]:to-[#00907c]">
   205→                                        <DollarSign className="w-4 h-4 text-[#00b14f] min-[1440px]:text-white" />
   206→                                    </div>
   207→                                    <div>
   208→                                        <p className="job-detail_info--section-content-title text-[12px] text-slate-400 font-medium min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#263a4d] min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.14px]">Quyền lợi / Lương</p>
   209→                                        <p className="job-detail_info--section-content-value text-[13px] text-[#212f3f] font-semibold truncate max-w-[180px] min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#212f3f] min-[1440px]:font-semibold min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.175px]" title={event.benefits || "Thỏa thuận"}>
   210→                                            {event.benefits || "Thỏa thuận"}
   211→                                        </p>
   212→                                    </div>
   213→                                </div>
   214→                                <div className="job-detail_info--section section-location flex items-center gap-3 min-[1440px]:flex-row min-[1440px]:items-center min-[1440px]:gap-[16px] min-[1440px]:flex-basis-0">
   215→                                    <div className="job-detail_info--section-icon w-9 h-9 rounded-full bg-[#00b14f]/5 flex items-center justify-center shrink-0 min-[1440px]:w-[40px] min-[1440px]:h-[40px] min-[1440px]:rounded-[30px] min-[1440px]:p-[10px] min-[1440px]:flex min-[1440px]:flex-col min-[1440px]:justify-center min-[1440px]:items-center min-[1440px]:gap-[10px] min-[1440px]:bg-gradient-to-br min-[1440px]:from-[#00bf5d] min-[1440px]:to-[#00907c]">
   216→                                        <MapPin className="w-4 h-4 text-[#00b14f] min-[1440px]:text-white" />
   217→                                    </div>
   218→                                    <div>
   219→                                        <p className="job-detail_info--section-content-title text-[12px] text-slate-400 font-medium min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#263a4d] min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.14px]">Khu vực</p>
   220→                                        <p className="job-detail_info--section-content-value text-[13px] text-[#212f3f] font-semibold truncate min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#212f3f] min-[1440px]:font-semibold min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.175px]" title={event.danang_wards?.name ? `P. ${event.danang_wards.name}` : "Đà Nẵng"}>
   221→                                            {event.danang_wards?.name ? `${event.danang_wards.name}` : "Đà Nẵng"}
   222→                                        </p>
   223→                                    </div>
   224→                                </div>
   225→                                <div className="job-detail_info--section section-eventdate flex items-center gap-3 min-[1440px]:flex-row min-[1440px]:items-center min-[1440px]:gap-[16px]">
   226→                                    <div className="job-detail_info--section-icon w-9 h-9 rounded-full bg-[#00b14f]/5 flex items-center justify-center shrink-0 min-[1440px]:w-[40px] min-[1440px]:h-[40px] min-[1440px]:rounded-[30px] min-[1440px]:p-[10px] min-[1440px]:flex min-[1440px]:flex-col min-[1440px]:justify-center min-[1440px]:items-center min-[1440px]:gap-[10px] min-[1440px]:bg-gradient-to-br min-[1440px]:from-[#00bf5d] min-[1440px]:to-[#00907c]">
   227→                                        <Calendar className="w-4 h-4 text-[#00b14f] min-[1440px]:text-white" />
   228→                                    </div>
   229→                                    <div>
   230→                                        <p className="job-detail_info--section-content-title text-[12px] text-slate-400 font-medium min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#263a4d] min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.14px]">Ngày diễn ra</p>
   231→                                        <p className="job-detail_info--section-content-value text-[13px] text-[#212f3f] font-semibold min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#212f3f] min-[1440px]:font-semibold min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.175px]">
   232→                                            {event.event_date ? new Date(event.event_date).toLocaleDateString('vi-VN') : "Đang cập nhật"}
   233→                                        </p>
   234→                                    </div>
   235→                                </div>
   236→                            </div>
   237→
   238→                            {/* Hạn chót nộp hồ sơ */}
   239→                            <div className="job-detail_info--flex flex items-center text-sm text-[#333] gap-2 mt-4 mb-6 min-[1440px]:w-[713px] min-[1440px]:h-[26px] min-[1440px]:flex min-[1440px]:items-center min-[1440px]:gap-[4px] min-[1440px]:text-[14px] min-[1440px]:leading-[20px] min-[1440px]:text-[#333] min-[1440px]:mt-[16px] min-[1440px]:mb-6">
   240→                                <div className="job-detail_info--deadline flex items-center gap-1 text-sm text-[#7f878f] bg-slate-50 px-2.5 py-1.5 md:p-[2px_8px_2px_4px] rounded min-[1440px]:rounded-[4px] min-[1440px]:gap-[4px] min-[1440px]:text-[14px] min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.14px] min-[1440px]:text-[#7f878f]">
   241→                                    <span>Hạn nộp hồ sơ</span>

   247→                                    <span className="deadline text-sm font-semibold text-[#263a4d] min-[1440px]:text-[14px] min-[1440px]:font-semibold min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.175px] min-[1440px]:text-[#263a4d]">

   255→                            </div>

   257→                            {/* Action Buttons Row */}
   258→                            <div className="job-detail_info--actions box-apply-current flex flex-wrap items-center gap-3 border-t border-slate-100 pt-5 min-[1440px]:w-[713px] min-[1440px]:h-[40px] min-[1440px]:flex min-[1440px]:items-center min-[1440px]:flex-wrap min-[1440px]:text-[14px] min-[1440px]:gap-[12px] min-[1440px]:leading-[20px] min-[1440px]:m-[4px_0px_0px] min-[1440px]:border-t-0 min-[1440px]:pt-0">
