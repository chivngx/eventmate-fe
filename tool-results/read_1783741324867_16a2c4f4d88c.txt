     1→"use client"
     2→
     3→import { useEffect, useState, useRef } from "react"
     4→import { useParams, useNavigate } from "@/lib/router"
     5→import { supabase } from "@/lib/supabase"
     6→import { getUserFacingMessage } from "@/lib/error"
     7→import MainLayout from "@/components/layout/MainLayout"
     8→import OrgLayout from "@/components/layout/OrgLayout"
     9→import { Send, MessageSquare, User, ArrowLeft, Calendar, Video, Clock, Check, X } from "lucide-react"
    10→import { Avatar, AvatarFallback, AvatarImage } from "@/components/ui/avatar"
    11→import { Button } from "@/components/ui/button"
    12→
    13→interface ChatItem {
    14→  id: string
    15→  event_id: string
    16→  student_id: string
    17→  organizer_id: string
    18→  created_at: string
    19→  events: { title: string }
    20→  student_profile: { id: string; full_name: string; avatar_url: string }
    21→  organizer_profile: { id: string; full_name: string; avatar_url: string }
    22→}
    23→
    24→interface Message {
    25→  id: string
    26→  chat_id: string
    27→  sender_id: string
    28→  content: string
    29→  created_at: string
    30→}
    31→
    32→export default function Chat() {
    33→  const { id: routeChatId } = useParams<{ id: string }>()
    34→  const navigate = useNavigate()
    35→
    36→  const [currentUser, setCurrentUser] = useState<any>(null)
    37→  const [role, setRole] = useState("student")
    38→  const [loading, setLoading] = useState(true)
    39→
    40→  const [chats, setChats] = useState<ChatItem[]>([])
    41→  const [activeChat, setActiveChat] = useState<ChatItem | null>(null)
    42→  const [messages, setMessages] = useState<Message[]>([])
    43→  const [newMessage, setNewMessage] = useState("")
    44→  const [sending, setSending] = useState(false)
    45→
    46→  // Phân trang
    47→  const [messagesLimit, setMessagesLimit] = useState(20)
    48→  const [hasMoreMessages, setHasMoreMessages] = useState(true)
    49→
    50→  // Hẹn phỏng vấn
    51→  const [isInterviewModalOpen, setIsInterviewModalOpen] = useState(false)
    52→  const [interviewTitle, setInterviewTitle] = useState("")
    53→  const [interviewDate, setInterviewDate] = useState("")
    54→  const [interviewLink, setInterviewLink] = useState("")
    55→  const [interviews, setInterviews] = useState<Record<string, any>>({})
    56→  const [creatingInterview, setCreatingInterview] = useState(false)
    57→
    58→  const messagesEndRef = useRef<HTMLDivElement>(null)
    59→
    60→  // 1. Lấy thông tin User hiện tại
    61→  useEffect(() => {
    62→    const fetchUser = async () => {
    63→      const { data: { user } } = await supabase.auth.getUser()
    64→      if (!user) {
    65→        navigate("/login")
    66→        return
    67→      }
    68→      setCurrentUser(user)
    69→
    70→      const { data: profile } = await supabase.from("profiles").select("role").eq("id", user.id).maybeSingle()
    71→      if (profile) setRole(profile.role)
    72→
    73→      setLoading(false)
    74→    }
    75→    fetchUser()
    76→  }, [navigate])
    77→
    78→  // 2. Fetch danh sách các phòng Chat của User
    79→  const fetchChats = async () => {
    80→    if (!currentUser) return
    81→
    82→    const isOrg = role === "organizer"
    83→    const queryField = isOrg ? "organizer_id" : "student_id"
    84→
    85→    const { data, error } = await supabase
    86→      .from("chats")
    87→      .select(`
    88→        id, event_id, student_id, organizer_id, created_at,
    89→        events (title),
    90→        student_profile:profiles!student_id (id, full_name, avatar_url),
    91→        organizer_profile:profiles!organizer_id (id, full_name, avatar_url)
    92→      `)
    93→      .eq(queryField, currentUser.id)
    94→      .order("created_at", { ascending: false })
    95→
    96→    if (error) {
    97→      console.error("Lỗi fetchChats:", error)
    98→    }
    99→
   100→    if (!error && data) {
   101→      const formattedChats = data.map((item: any) => ({
   102→        ...item,
   103→        events: Array.isArray(item.events) ? item.events[0] || { title: "Sự kiện chung" } : item.events || { title: "Sự kiện chung" },
   104→        student_profile: Array.isArray(item.student_profile) ? item.student_profile[0] || { id: "", full_name: "Ứng viên", avatar_url: "" } : item.student_profile || { id: "", full_name: "Ứng viên", avatar_url: "" },
   105→        organizer_profile: Array.isArray(item.organizer_profile) ? item.organizer_profile[0] || { id: "", full_name: "Ban tổ chức", avatar_url: "" } : item.organizer_profile || { id: "", full_name: "Ban tổ chức", avatar_url: "" }
   106→      })) as ChatItem[]
   107→
   108→      setChats(formattedChats)
   109→
   110→      // Nếu có chat ID từ Route, set active chat
   111→      if (routeChatId) {
   112→        const found = formattedChats.find(c => c.id === routeChatId)
   113→        if (found) {
   114→          setActiveChat(found)
   115→        } else {
   116→          // Trực tiếp fetch chat nếu không tìm thấy trong list (phòng ngừa lỗi đồng bộ)
   117→          const { data: directChat } = await supabase
   118→            .from("chats")
   119→            .select(`
   120→              id, event_id, student_id, organizer_id, created_at,
   121→              events (title),
   122→              student_profile:profiles!student_id (id, full_name, avatar_url),
   123→              organizer_profile:profiles!organizer_id (id, full_name, avatar_url)
   124→            `)
   125→            .eq("id", routeChatId)
   126→            .maybeSingle()
   127→
   128→          if (directChat) {
   129→            const formatted = {
   130→              ...directChat,
   131→              events: Array.isArray(directChat.events) ? directChat.events[0] || { title: "Sự kiện chung" } : directChat.events || { title: "Sự kiện chung" },
   132→              student_profile: Array.isArray(directChat.student_profile) ? directChat.student_profile[0] || { id: "", full_name: "Ứng viên", avatar_url: "" } : directChat.student_profile || { id: "", full_name: "Ứng viên", avatar_url: "" },
   133→              organizer_profile: Array.isArray(directChat.organizer_profile) ? directChat.organizer_profile[0] || { id: "", full_name: "Ban tổ chức", avatar_url: "" } : directChat.organizer_profile || { id: "", full_name: "Ban tổ chức", avatar_url: "" }
   134→            } as ChatItem
   135→            setActiveChat(formatted)
   136→          }
   137→        }
   138→      } else if (formattedChats.length > 0 && !activeChat) {
   139→        setActiveChat(formattedChats[0])
   140→      }
   141→    }
   142→  }
   143→
   144→  useEffect(() => {
   145→    if (currentUser) {
   146→      fetchChats()
   147→    }
   148→  }, [currentUser, role, routeChatId])
   149→
   150→  // Fetch danh sách lịch hẹn phỏng vấn
   151→  const fetchInterviews = async (chat: ChatItem) => {
   152→    const { data, error } = await supabase
   153→      .from("interviews")
   154→      .select("*")
   155→      .eq("student_id", chat.student_id)
   156→      .eq("organizer_id", chat.organizer_id)
   157→
   158→    if (!error && data) {
   159→      const map: Record<string, any> = {}
   160→      data.forEach((item) => {
   161→        map[item.id] = item
   162→      })
   163→      setInterviews(map)
   164→    }
   165→  }
   166→
   167→  // 3. Fetch tin nhắn khi đổi Chat room
   168→  const fetchMessages = async (chatId: string, currentLimit = 20) => {
   169→    const { data, error } = await supabase
   170→      .from("messages")
   171→      .select("*")
   172→      .eq("chat_id", chatId)
   173→      .order("created_at", { ascending: false })
   174→      .limit(currentLimit)
   175→
   176→    if (!error && data) {
   177→      const reversed = [...data].reverse()
   178→      setMessages(reversed)
   179→      setHasMoreMessages(data.length === currentLimit)
   180→    }
   181→  }
   182→
   183→  useEffect(() => {
   184→    if (activeChat) {
   185→      setMessagesLimit(20)
   186→      fetchMessages(activeChat.id, 20)
   187→      fetchInterviews(activeChat)
   188→
   189→      // 4. Đăng ký Realtime tin nhắn mới của Chat room đang mở
   190→      const channel = supabase
   191→        .channel(`chat-room-${activeChat.id}`)
   192→        .on(
   193→          "postgres_changes",
   194→          {
   195→            event: "INSERT",
   196→            schema: "public",
   197→            table: "messages"
   198→          },
   199→          (payload) => {
   200→            const newMsg = payload.new as Message
   201→            if (newMsg.chat_id === activeChat.id) {
   202→              setMessages((prev) => {
   203→                if (prev.some((m) => m.id === newMsg.id)) return prev
   204→                return [...prev, newMsg]
   205→              })
   206→
   207→              // Nếu là tin nhắn lịch hẹn mới, tải lại danh sách lịch hẹn
   208→              if (newMsg.content.startsWith("__INTERVIEW_REQUEST__:")) {
   209→                fetchInterviews(activeChat)
   210→              }
   211→
   212→              scrollToBottom()
   213→            }
   214→          }
   215→        )
   216→        .subscribe()
   217→
   218→      return () => {
   219→        supabase.removeChannel(channel)
   220→      }
   221→    }
   222→  }, [activeChat])
   223→
   224→  // Polling fallback to fetch messages every 4 seconds (handles cases where Supabase Realtime replication is disabled)
   225→  useEffect(() => {
   226→    if (!activeChat) return
   227→
   228→    const interval = setInterval(() => {
   229→      fetchMessages(activeChat.id, messagesLimit)
   230→      fetchInterviews(activeChat)
   231→    }, 4000)
   232→
   233→    return () => clearInterval(interval)
   234→  }, [activeChat, messagesLimit])
   235→
   236→  const scrollToBottom = () => {
   237→    setTimeout(() => {
   238→      messagesEndRef.current?.scrollIntoView({ behavior: "smooth" })
   239→    }, 100)
   240→  }
   241→
   242→  // 5. Gửi tin nhắn
   243→  const handleSendMessage = async (e: React.FormEvent) => {
   244→    e.preventDefault()
   245→    if (!newMessage.trim() || !activeChat || !currentUser || sending) return
   246→
   247→    setSending(true)
   248→    const tempMsg = newMessage
   249→    setNewMessage("")
   250→
   251→    // Lưu tin nhắn gửi đi và lấy dữ liệu trả về để cập nhật UI ngay lập tức
   252→    const { data, error } = await supabase
   253→      .from("messages")
   254→      .insert([
   255→        {
   256→          chat_id: activeChat.id,
   257→          sender_id: currentUser.id,
   258→          content: tempMsg
   259→        }
   260→      ])
   261→      .select()
   262→      .single()
   263→
   264→    setSending(false)
   265→
   266→    if (error) {
   267→      console.error("Lỗi gửi tin nhắn:", error)
   268→      setNewMessage(tempMsg) // Khôi phục lại text
   269→    } else if (data) {
   270→      setMessages((prev) => {
   271→        if (prev.some((m) => m.id === data.id)) return prev
   272→        return [...prev, data as Message]
   273→      })
   274→      scrollToBottom()
   275→    }
   276→  }
   277→
   278→  // Tạo lịch phỏng vấn
   279→  const handleCreateInterview = async (e: React.FormEvent) => {
   280→    e.preventDefault()
   281→    if (!activeChat || !currentUser || !interviewTitle || !interviewDate || creatingInterview) return
   282→
   283→    setCreatingInterview(true)
   284→    try {
   285→      // 1. Tạo bản ghi lịch hẹn
   286→      const { data: interview, error: intError } = await supabase
   287→        .from("interviews")
   288→        .insert([
   289→          {
   290→            event_id: activeChat.event_id,
   291→            organizer_id: currentUser.id,
   292→            student_id: activeChat.student_id,
   293→            title: interviewTitle,
   294→            scheduled_at: new Date(interviewDate).toISOString(),
   295→            meeting_link: interviewLink || null,
   296→            status: "pending"
   297→          }
   298→        ])
   299→        .select()
   300→        .single()
   301→
   302→      if (intError) throw intError
   303→
   304→      // 2. Gửi tin nhắn đặc biệt vào chat room
   305→      const { data: messageData, error: msgError } = await supabase
   306→        .from("messages")
   307→        .insert([
   308→          {
   309→            chat_id: activeChat.id,
   310→            sender_id: currentUser.id,
   311→            content: `__INTERVIEW_REQUEST__:${interview.id}`
   312→          }
   313→        ])
   314→        .select()
   315→        .single()
   316→
   317→      if (msgError) throw msgError
   318→
   319→      // Cập nhật local state
   320→      setInterviews(prev => ({ ...prev, [interview.id]: interview }))
   321→      if (messageData) {
   322→        setMessages(prev => [...prev, messageData as Message])
   323→      }
   324→
   325→      // Reset form
   326→      setInterviewTitle("")
   327→      setInterviewDate("")
   328→      setInterviewLink("")
   329→      setIsInterviewModalOpen(false)
   330→      scrollToBottom()
   331→    } catch (err: any) {
   332→      console.error("Lỗi tạo lịch hẹn:", err)
   333→      alert(getUserFacingMessage(err, "Không thể tạo lịch hẹn. Vui lòng thử lại."))
   334→    } finally {
   335→      setCreatingInterview(false)
   336→    }
   337→  }
   338→
   339→  // Chấp nhận/Từ chối phỏng vấn
   340→  const handleUpdateInterviewStatus = async (interviewId: string, newStatus: "accepted" | "rejected") => {
   341→    try {
   342→      const interview = interviews[interviewId]
   343→      if (!interview) return
   344→
   345→      const { error } = await supabase
   346→        .from("interviews")
   347→        .update({ status: newStatus })
   348→        .eq("id", interviewId)
   349→
   350→      if (error) throw error
   351→
   352→      // Cập nhật state
   353→      setInterviews(prev => ({
   354→        ...prev,
   355→        [interviewId]: { ...prev[interviewId], status: newStatus }
   356→      }))
   357→
   358→      // Gửi thông báo đến organizer
   359→      await supabase.from("notifications").insert([
   360→        {
   361→          user_id: interview.organizer_id,
   362→          title: newStatus === "accepted" ? "Lịch phỏng vấn được chấp nhận" : "Lịch phỏng vấn bị từ chối",
   363→          message: `Ứng viên đã ${newStatus === "accepted" ? "chấp nhận" : "từ chối"} lịch hẹn phỏng vấn: ${interview.title}`,
   364→          is_read: false
   365→        }
   366→      ])
   367→    } catch (err: any) {
   368→      console.error("Lỗi cập nhật lịch hẹn:", err)
   369→      alert(getUserFacingMessage(err, "Không thể cập nhật trạng thái. Vui lòng thử lại."))
   370→    }
   371→  }
   372→
   373→  if (loading) {
   374→    return (
   375→      <div className="flex h-screen w-screen items-center justify-center bg-slate-50 dark:bg-slate-950">
   376→        <div className="h-8 w-8 animate-spin rounded-full border-4 border-emerald-600 border-t-transparent"></div>
   377→      </div>
   378→    )
   379→  }
   380→
   381→  const getPartnerProfile = (chat: ChatItem) => {
   382→    return role === "organizer" ? chat.student_profile : chat.organizer_profile
   383→  }
   384→
   385→  const chatUI = (
   386→    <>
   387→      <div className="max-w-6xl mx-auto h-[calc(100vh-120px)] flex bg-white dark:bg-slate-900 rounded-[2.5rem] border-2 border-slate-100 dark:border-slate-800/80 shadow-2xl overflow-hidden">
   388→        {/* Left Side: Danh sách Chats */}
   389→        <div className={`w-full md:w-80 border-r-2 border-slate-100 dark:border-slate-850 flex flex-col ${activeChat ? "hidden md:flex" : "flex"}`}>
   390→          <div className="p-5 border-b-2 border-slate-100 dark:border-slate-850">
   391→            <h1 className="text-xl font-black text-slate-900 dark:text-slate-100 flex items-center gap-2">
   392→              <MessageSquare className="w-5 h-5 text-emerald-500" />
   393→              Hộp thư trò chuyện
   394→            </h1>
   395→          </div>
   396→          <div className="flex-1 overflow-y-auto p-3 space-y-1.5">
   397→            {chats.length === 0 ? (
   398→              <div className="text-center py-12 text-slate-400 dark:text-slate-500 font-semibold text-xs">
   399→                Chưa có hội thoại nào
   400→              </div>
   401→            ) : (
   402→              chats.map((chat) => {
   403→                const partner = getPartnerProfile(chat)
   404→                const isActive = activeChat?.id === chat.id
   405→                return (
   406→                  <button
   407→                    key={chat.id}
   408→                    onClick={() => {
   409→                      setActiveChat(chat)
   410→                      navigate(`/chat/${chat.id}`)
   411→                    }}
   412→                    className={`w-full flex items-center gap-3 p-3.5 rounded-2xl transition-all text-left ${isActive
   413→                        ? "bg-emerald-50 dark:bg-emerald-950/20 border border-emerald-100/50 dark:border-emerald-900/30"
   414→                        : "hover:bg-slate-50 dark:hover:bg-slate-800/40 border border-transparent"
   415→                      }`}
   416→                  >
   417→                    <Avatar className="h-10 w-10 border border-white dark:border-slate-800 shadow-sm shrink-0">
   418→                      <AvatarImage src={partner.avatar_url} />
   419→                      <AvatarFallback className="bg-emerald-100 text-emerald-700 font-bold">
   420→                        {partner.full_name?.charAt(0).toUpperCase() || <User className="w-4 h-4" />}
   421→                      </AvatarFallback>
   422→                    </Avatar>
   423→                    <div className="min-w-0 flex-1">
   424→                      <h4 className={`text-sm font-bold truncate ${isActive ? "text-emerald-700 dark:text-emerald-400" : "text-slate-800 dark:text-slate-200"}`}>
   425→                        {partner.full_name}
   426→                      </h4>
   427→                      <p className="text-[11px] text-slate-400 dark:text-slate-500 truncate font-semibold">
   428→                        Sự kiện: {chat.events.title}
   429→                      </p>
   430→                    </div>
   431→                  </button>
   432→                )
   433→              })
   434→            )}
   435→          </div>
   436→        </div>
   437→
   438→        {/* Right Side: Message Thread */}
   439→        <div className={`flex-1 flex flex-col bg-slate-50/50 dark:bg-slate-900/20 ${!activeChat ? "hidden md:flex items-center justify-center" : "flex"}`}>
   440→          {activeChat ? (
   441→            <>
   442→              {/* Header */}
   443→              <div className="bg-white dark:bg-slate-900 p-4 border-b-2 border-slate-100 dark:border-slate-850 flex items-center justify-between">
   444→                <div className="flex items-center gap-3 min-w-0">
   445→                  <button
   446→                    onClick={() => {
   447→                      setActiveChat(null)
   448→                      navigate("/chat")
   449→                    }}
   450→                    className="md:hidden p-1.5 hover:bg-slate-100 dark:hover:bg-slate-800 rounded-xl text-slate-400 hover:text-slate-600 dark:hover:text-slate-250 transition-colors mr-1"
   451→                  >
   452→                    <ArrowLeft className="w-5 h-5" />
   453→                  </button>
   454→                  <Avatar className="h-10 w-10 border shadow-inner shrink-0">
   455→                    <AvatarImage src={getPartnerProfile(activeChat).avatar_url} />
   456→                    <AvatarFallback className="bg-emerald-100 text-emerald-700 font-black">
   457→                      {getPartnerProfile(activeChat).full_name?.charAt(0).toUpperCase() || <User className="w-4 h-4" />}
   458→                    </AvatarFallback>
   459→                  </Avatar>
   460→                  <div className="min-w-0">
   461→                    <h3 className="text-sm font-extrabold text-slate-900 dark:text-slate-100 truncate">
   462→                      {getPartnerProfile(activeChat).full_name}
   463→                    </h3>
   464→                    <p className="text-[10px] font-bold text-slate-400 dark:text-slate-500 truncate">
   465→                      Đang liên hệ về: {activeChat.events.title}
   466→                    </p>
   467→                  </div>
   468→                </div>
   469→
   470→                {role === "organizer" && (
   471→                  <Button
   472→                    onClick={() => setIsInterviewModalOpen(true)}
   473→                    className="bg-emerald-600 hover:bg-emerald-700 text-white font-bold text-xs rounded-xl h-9 px-4 flex items-center gap-1.5 shadow-sm transition-colors"
   474→                  >
   475→                    <Calendar className="w-3.5 h-3.5" />
   476→                    Hẹn phỏng vấn
   477→                  </Button>
   478→                )}
   479→              </div>
   480→
   481→              {/* Messages Body */}
   482→              <div className="flex-1 overflow-y-auto p-5 space-y-3.5">
   483→                {/* Nút tải thêm tin nhắn cũ */}
   484→                {hasMoreMessages && (
   485→                  <div className="flex justify-center pb-3">
   486→                    <button
   487→                      onClick={() => {
   488→                        const nextLimit = messagesLimit + 20
   489→                        setMessagesLimit(nextLimit)
   490→                        fetchMessages(activeChat.id, nextLimit)
   491→                      }}
   492→                      className="text-[11px] font-black text-emerald-600 dark:text-emerald-400 bg-emerald-50 dark:bg-emerald-950/20 px-3.5 py-1.5 rounded-full hover:bg-emerald-100 dark:hover:bg-emerald-950/40 transition-colors"
   493→                    >
   494→                      Tải tin nhắn cũ hơn
   495→                    </button>
   496→                  </div>
   497→                )}
   498→
   499→                {messages.map((msg) => {
   500→                  const isMe = msg.sender_id === currentUser.id
   501→
   502→                  // Kiểm tra xem tin nhắn có phải là lời mời phỏng vấn không
   503→                  if (msg.content.startsWith("__INTERVIEW_REQUEST__:")) {
   504→                    const interviewId = msg.content.substring("__INTERVIEW_REQUEST__:".length)
   505→                    const interview = interviews[interviewId]
   506→
   507→                    if (!interview) {
   508→                      return (
   509→                        <div key={msg.id} className="flex justify-center my-2">
