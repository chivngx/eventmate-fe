     1→     1→"use client"
     2→     2→
     3→     3→import { useEffect, useState, useRef } from "react"
     4→     4→import { useParams, useNavigate } from "@/lib/router"
     5→     5→import { supabase } from "@/lib/supabase"
     6→     6→import { getUserFacingMessage } from "@/lib/error"
     7→     7→import MainLayout from "@/components/layout/MainLayout"
     8→     8→import OrgLayout from "@/components/layout/OrgLayout"
     9→     9→import { Send, MessageSquare, User, ArrowLeft, Calendar, Video, Clock, Check, X } from "lucide-react"
    10→    10→import { Avatar, AvatarFallback, AvatarImage } from "@/components/ui/avatar"
    11→    11→import { Button } from "@/components/ui/button"
    12→    12→
    13→    13→interface ChatItem {
    14→    14→  id: string
    15→    15→  event_id: string
    16→    16→  student_id: string
    17→    17→  organizer_id: string
    18→    18→  created_at: string
    19→    19→  events: { title: string }
    20→    20→  student_profile: { id: string; full_name: string; avatar_url: string }
    21→    21→  organizer_profile: { id: string; full_name: string; avatar_url: string }
    22→    22→}
    23→    23→
    24→    24→interface Message {
    25→    25→  id: string
    26→    26→  chat_id: string
    27→    27→  sender_id: string
    28→    28→  content: string
    29→    29→  created_at: string
    30→    30→}
    31→    31→
    32→    32→export default function Chat() {
    33→    33→  const { id: routeChatId } = useParams<{ id: string }>()
    34→    34→  const navigate = useNavigate()
    35→    35→
    36→    36→  const [currentUser, setCurrentUser] = useState<any>(null)
    37→    37→  const [role, setRole] = useState("student")
    38→    38→  const [loading, setLoading] = useState(true)
    39→    39→
    40→    40→  const [chats, setChats] = useState<ChatItem[]>([])
    41→    41→  const [activeChat, setActiveChat] = useState<ChatItem | null>(null)
    42→    42→  const [messages, setMessages] = useState<Message[]>([])
    43→    43→  const [newMessage, setNewMessage] = useState("")
    44→    44→  const [sending, setSending] = useState(false)
    45→    45→
    46→    46→  // Phân trang
    47→    47→  const [messagesLimit, setMessagesLimit] = useState(20)
    48→    48→  const [hasMoreMessages, setHasMoreMessages] = useState(true)
    49→    49→
    50→    50→  // Hẹn phỏng vấn
    51→    51→  const [isInterviewModalOpen, setIsInterviewModalOpen] = useState(false)
    52→    52→  const [interviewTitle, setInterviewTitle] = useState("")
    53→    53→  const [interviewDate, setInterviewDate] = useState("")
    54→    54→  const [interviewLink, setInterviewLink] = useState("")
    55→    55→  const [interviews, setInterviews] = useState<Record<string, any>>({})
    56→    56→  const [creatingInterview, setCreatingInterview] = useState(false)
    57→    57→
    58→    58→  const messagesEndRef = useRef<HTMLDivElement>(null)
    59→    59→
    60→    60→  // 1. Lấy thông tin User hiện tại
    61→    61→  useEffect(() => {
    62→    62→    const fetchUser = async () => {
    63→    63→      const { data: { user } } = await supabase.auth.getUser()
    64→    64→      if (!user) {
    65→    65→        navigate("/login")
    66→    66→        return
    67→    67→      }
    68→    68→      setCurrentUser(user)
    69→    69→
    70→    70→      const { data: profile } = await supabase.from("profiles").select("role").eq("id", user.id).maybeSingle()
    71→    71→      if (profile) setRole(profile.role)
    72→    72→
    73→    73→      setLoading(false)
    74→    74→    }
    75→    75→    fetchUser()
    76→    76→  }, [navigate])
    77→    77→
    78→    78→  // 2. Fetch danh sách các phòng Chat của User
    79→    79→  const fetchChats = async () => {
    80→    80→    if (!currentUser) return
    81→    81→
    82→    82→    const isOrg = role === "organizer"
    83→    83→    const queryField = isOrg ? "organizer_id" : "student_id"
    84→    84→
    85→    85→    const { data, error } = await supabase
    86→    86→      .from("chats")
    87→    87→      .select(`
    88→    88→        id, event_id, student_id, organizer_id, created_at,
    89→    89→        events (title),
    90→    90→        student_profile:profiles!student_id (id, full_name, avatar_url),
    91→    91→        organizer_profile:profiles!organizer_id (id, full_name, avatar_url)
    92→    92→      `)
    93→    93→      .eq(queryField, currentUser.id)
    94→    94→      .order("created_at", { ascending: false })
    95→    95→
    96→    96→    if (error) {
    97→    97→      console.error("Lỗi fetchChats:", error)
    98→    98→    }
    99→    99→
   100→   100→    if (!error && data) {
   101→   101→      const formattedChats = data.map((item: any) => ({
   102→   102→        ...item,
   103→   103→        events: Array.isArray(item.events) ? item.events[0] || { title: "Sự kiện chung" } : item.events || { title: "Sự kiện chung" },
   104→   104→        student_profile: Array.isArray(item.student_profile) ? item.student_profile[0] || { id: "", full_name: "Ứng viên", avatar_url: "" } : item.student_profile || { id: "", full_name: "Ứng viên", avatar_url: "" },
   105→   105→        organizer_profile: Array.isArray(item.organizer_profile) ? item.organizer_profile[0] || { id: "", full_name: "Ban tổ chức", avatar_url: "" } : item.organizer_profile || { id: "", full_name: "Ban tổ chức", avatar_url: "" }
   106→   106→      })) as ChatItem[]
   107→   107→
   108→   108→      setChats(formattedChats)
   109→   109→
   110→   110→      // Nếu có chat ID từ Route, set active chat
   111→   111→      if (routeChatId) {
   112→   112→        const found = formattedChats.find(c => c.id === routeChatId)
   113→   113→        if (found) {
   114→   114→          setActiveChat(found)
   115→   115→        } else {
   116→   116→          // Trực tiếp fetch chat nếu không tìm thấy trong list (phòng ngừa lỗi đồng bộ)
   117→   117→          const { data: directChat } = await supabase
   118→   118→            .from("chats")
   119→   119→            .select(`
   120→   120→              id, event_id, student_id, organizer_id, created_at,
   121→   121→              events (title),
   122→   122→              student_profile:profiles!student_id (id, full_name, avatar_url),
   123→   123→              organizer_profile:profiles!organizer_id (id, full_name, avatar_url)
   124→   124→            `)
   125→   125→            .eq("id", routeChatId)
   126→   126→            .maybeSingle()
   127→   127→
   128→   128→          if (directChat) {
   129→   129→            const formatted = {
   130→   130→              ...directChat,
   131→   131→              events: Array.isArray(directChat.events) ? directChat.events[0] || { title: "Sự kiện chung" } : directChat.events || { title: "Sự kiện chung" },
   132→   132→              student_profile: Array.isArray(directChat.student_profile) ? directChat.student_profile[0] || { id: "", full_name: "Ứng viên", avatar_url: "" } : directChat.student_profile || { id: "", full_name: "Ứng viên", avatar_url: "" },
   133→   133→              organizer_profile: Array.isArray(directChat.organizer_profile) ? directChat.organizer_profile[0] || { id: "", full_name: "Ban tổ chức", avatar_url: "" } : directChat.organizer_profile || { id: "", full_name: "Ban tổ chức", avatar_url: "" }
   134→   134→            } as ChatItem
   135→   135→            setActiveChat(formatted)
   136→   136→          }
   137→   137→        }
   138→   138→      } else if (formattedChats.length > 0 && !activeChat) {
   139→   139→        setActiveChat(formattedChats[0])
   140→   140→      }
   141→   141→    }
   142→   142→  }
   143→   143→
   144→   144→  useEffect(() => {
   145→   145→    if (currentUser) {
   146→   146→      fetchChats()
   147→   147→    }
   148→   148→  }, [currentUser, role, routeChatId])
   149→   149→
   150→   150→  // Fetch danh sách lịch hẹn phỏng vấn
   151→   151→  const fetchInterviews = async (chat: ChatItem) => {
   152→   152→    const { data, error } = await supabase
   153→   153→      .from("interviews")
   154→   154→      .select("*")
   155→   155→      .eq("student_id", chat.student_id)
   156→   156→      .eq("organizer_id", chat.organizer_id)
   157→   157→
   158→   158→    if (!error && data) {
   159→   159→      const map: Record<string, any> = {}
   160→   160→      data.forEach((item) => {
   161→   161→        map[item.id] = item
   162→   162→      })
   163→   163→      setInterviews(map)
   164→   164→    }
   165→   165→  }
   166→   166→
   167→   167→  // 3. Fetch tin nhắn khi đổi Chat room
   168→   168→  const fetchMessages = async (chatId: string, currentLimit = 20) => {
   169→   169→    const { data, error } = await supabase
   170→   170→      .from("messages")
   171→   171→      .select("*")
   172→   172→      .eq("chat_id", chatId)
   173→   173→      .order("created_at", { ascending: false })
   174→   174→      .limit(currentLimit)
   175→   175→
   176→   176→    if (!error && data) {
   177→   177→      const reversed = [...data].reverse()
   178→   178→      setMessages(reversed)
   179→   179→      setHasMoreMessages(data.length === currentLimit)
   180→   180→    }
   181→   181→  }
   182→   182→
   183→   183→  useEffect(() => {
   184→   184→    if (activeChat) {
   185→   185→      setMessagesLimit(20)
   186→   186→      fetchMessages(activeChat.id, 20)
   187→   187→      fetchInterviews(activeChat)
   188→   188→
   189→   189→      // 4. Đăng ký Realtime tin nhắn mới của Chat room đang mở
   190→   190→      const channel = supabase
   191→   191→        .channel(`chat-room-${activeChat.id}`)
   192→   192→        .on(
   193→   193→          "postgres_changes",
   194→   194→          {
   195→   195→            event: "INSERT",
   196→   196→            schema: "public",
   197→   197→            table: "messages"
   198→   198→          },
   199→   199→          (payload) => {
   200→   200→            const newMsg = payload.new as Message
   201→   201→            if (newMsg.chat_id === activeChat.id) {
   202→   202→              setMessages((prev) => {
   203→   203→                if (prev.some((m) => m.id === newMsg.id)) return prev
   204→   204→                return [...prev, newMsg]
   205→   205→              })
   206→   206→
   207→   207→              // Nếu là tin nhắn lịch hẹn mới, tải lại danh sách lịch hẹn
   208→   208→              if (newMsg.content.startsWith("__INTERVIEW_REQUEST__:")) {
   209→   209→                fetchInterviews(activeChat)
   210→   210→              }
   211→   211→
   212→   212→              scrollToBottom()
   213→   213→            }
   214→   214→          }
   215→   215→        )
   216→   216→        .subscribe()
   217→   217→
   218→   218→      return () => {
   219→   219→        supabase.removeChannel(channel)
   220→   220→      }
   221→   221→    }
   222→   222→  }, [activeChat])
   223→   223→
   224→   224→  // Polling fallback to fetch messages every 4 seconds (handles cases where Supabase Realtime replication is disabled)
   225→   225→  useEffect(() => {
   226→   226→    if (!activeChat) return
   227→   227→
   228→   228→    const interval = setInterval(() => {
   229→   229→      fetchMessages(activeChat.id, messagesLimit)
   230→   230→      fetchInterviews(activeChat)
   231→   231→    }, 4000)
   232→   232→
   233→   233→    return () => clearInterval(interval)
   234→   234→  }, [activeChat, messagesLimit])
   235→   235→
   236→   236→  const scrollToBottom = () => {
   237→   237→    setTimeout(() => {
   238→   238→      messagesEndRef.current?.scrollIntoView({ behavior: "smooth" })
   239→   239→    }, 100)
   240→   240→  }
   241→   241→
   242→   242→  // 5. Gửi tin nhắn
   243→   243→  const handleSendMessage = async (e: React.FormEvent) => {
   244→   244→    e.preventDefault()
   245→   245→    if (!newMessage.trim() || !activeChat || !currentUser || sending) return
   246→   246→
   247→   247→    setSending(true)
   248→   248→    const tempMsg = newMessage
   249→   249→    setNewMessage("")
   250→   250→
   251→   251→    // Lưu tin nhắn gửi đi và lấy dữ liệu trả về để cập nhật UI ngay lập tức
   252→   252→    const { data, error } = await supabase
   253→   253→      .from("messages")
   254→   254→      .insert([
   255→   255→        {
   256→   256→          chat_id: activeChat.id,
   257→   257→          sender_id: currentUser.id,
   258→   258→          content: tempMsg
   259→   259→        }
   260→   260→      ])
   261→   261→      .select()
   262→   262→      .single()
   263→   263→
   264→   264→    setSending(false)
   265→   265→
   266→   266→    if (error) {
   267→   267→      console.error("Lỗi gửi tin nhắn:", error)
   268→   268→      setNewMessage(tempMsg) // Khôi phục lại text
   269→   269→    } else if (data) {
   270→   270→      setMessages((prev) => {
   271→   271→        if (prev.some((m) => m.id === data.id)) return prev
   272→   272→        return [...prev, data as Message]
   273→   273→      })
   274→   274→      scrollToBottom()
   275→   275→    }
   276→   276→  }
   277→   277→
   278→   278→  // Tạo lịch phỏng vấn
   279→   279→  const handleCreateInterview = async (e: React.FormEvent) => {
   280→   280→    e.preventDefault()
   281→   281→    if (!activeChat || !currentUser || !interviewTitle || !interviewDate || creatingInterview) return
   282→   282→
   283→   283→    setCreatingInterview(true)
   284→   284→    try {
   285→   285→      // 1. Tạo bản ghi lịch hẹn
   286→   286→      const { data: interview, error: intError } = await supabase
   287→   287→        .from("interviews")
   288→   288→        .insert([
   289→   289→          {
   290→   290→            event_id: activeChat.event_id,
   291→   291→            organizer_id: currentUser.id,
   292→   292→            student_id: activeChat.student_id,
   293→   293→            title: interviewTitle,
   294→   294→            scheduled_at: new Date(interviewDate).toISOString(),
   295→   295→            meeting_link: interviewLink || null,
   296→   296→            status: "pending"
   297→   297→          }
   298→   298→        ])
   299→   299→        .select()
   300→   300→        .single()
   301→   301→
   302→   302→      if (intError) throw intError
   303→   303→
   304→   304→      // 2. Gửi tin nhắn đặc biệt vào chat room
   305→   305→      const { data: messageData, error: msgError } = await supabase
   306→   306→        .from("messages")
   307→   307→        .insert([
   308→   308→          {
   309→   309→            chat_id: activeChat.id,
   310→   310→            sender_id: currentUser.id,
   311→   311→            content: `__INTERVIEW_REQUEST__:${interview.id}`
   312→   312→          }
   313→   313→        ])
   314→   314→        .select()
   315→   315→        .single()
   316→   316→
   317→   317→      if (msgError) throw msgError
   318→   318→
   319→   319→      // Cập nhật local state
   320→   320→      setInterviews(prev => ({ ...prev, [interview.id]: interview }))
   321→   321→      if (messageData) {
   322→   322→        setMessages(prev => [...prev, messageData as Message])
   323→   323→      }
   324→   324→
   325→   325→      // Reset form
   326→   326→      setInterviewTitle("")
   327→   327→      setInterviewDate("")
   328→   328→      setInterviewLink("")
   329→   329→      setIsInterviewModalOpen(false)
   330→   330→      scrollToBottom()
   331→   331→    } catch (err: any) {
   332→   332→      console.error("Lỗi tạo lịch hẹn:", err)
   333→   333→      alert(getUserFacingMessage(err, "Không thể tạo lịch hẹn. Vui lòng thử lại."))
   334→   334→    } finally {
   335→   335→      setCreatingInterview(false)
   336→   336→    }
   337→   337→  }
   338→   338→
   339→   339→  // Chấp nhận/Từ chối phỏng vấn
   340→   340→  const handleUpdateInterviewStatus = async (interviewId: string, newStatus: "accepted" | "rejected") => {
   341→   341→    try {
   342→   342→      const interview = interviews[interviewId]
   343→   343→      if (!interview) return
   344→   344→
   345→   345→      const { error } = await supabase
   346→   346→        .from("interviews")
   347→   347→        .update({ status: newStatus })
   348→   348→        .eq("id", interviewId)
   349→   349→
   350→   350→      if (error) throw error
   351→   351→
   352→   352→      // Cập nhật state
   353→   353→      setInterviews(prev => ({
   354→   354→        ...prev,
   355→   355→        [interviewId]: { ...prev[interviewId], status: newStatus }
   356→   356→      }))
   357→   357→
   358→   358→      // Gửi thông báo đến organizer
   359→   359→      await supabase.from("notifications").insert([
   360→   360→        {
   361→   361→          user_id: interview.organizer_id,
   362→   362→          title: newStatus === "accepted" ? "Lịch phỏng vấn được chấp nhận" : "Lịch phỏng vấn bị từ chối",
   363→   363→          message: `Ứng viên đã ${newStatus === "accepted" ? "chấp nhận" : "từ chối"} lịch hẹn phỏng vấn: ${interview.title}`,
   364→   364→          is_read: false
   365→   365→        }
   366→   366→      ])
   367→   367→    } catch (err: any) {
   368→   368→      console.error("Lỗi cập nhật lịch hẹn:", err)
   369→   369→      alert(getUserFacingMessage(err, "Không thể cập nhật trạng thái. Vui lòng thử lại."))
   370→   370→    }
   371→   371→  }
   372→   372→
   373→   373→  if (loading) {
   374→   374→    return (
   375→   375→      <div className="flex h-screen w-screen items-center justify-center bg-slate-50 dark:bg-slate-950">
   376→   376→        <div className="h-8 w-8 animate-spin rounded-full border-4 border-emerald-600 border-t-transparent"></div>
   377→   377→      </div>
   378→   378→    )
   379→   379→  }
   380→   380→
   381→   381→  const getPartnerProfile = (chat: ChatItem) => {
   382→   382→    return role === "organizer" ? chat.student_profile : chat.organizer_profile
   383→   383→  }
   384→   384→
   385→   385→  const chatUI = (
   386→   386→    <>
   387→   387→      <div className="max-w-6xl mx-auto h-[calc(100vh-120px)] flex bg-white dark:bg-slate-900 rounded-[2.5rem] border-2 border-slate-100 dark:border-slate-800/80 shadow-2xl overflow-hidden">
   388→   388→        {/* Left Side: Danh sách Chats */}
   389→   389→        <div className={`w-full md:w-80 border-r-2 border-slate-100 dark:border-slate-850 flex flex-col ${activeChat ? "hidden md:flex" : "flex"}`}>
   390→   390→          <div className="p-5 border-b-2 border-slate-100 dark:border-slate-850">
   391→   391→            <h1 className="text-xl font-black text-slate-900 dark:text-slate-100 flex items-center gap-2">
   392→   392→              <MessageSquare className="w-5 h-5 text-emerald-500" />
   393→   393→              Hộp thư trò chuyện
   394→   394→            </h1>
   395→   395→          </div>
   396→   396→          <div className="flex-1 overflow-y-auto p-3 space-y-1.5">
   397→   397→            {chats.length === 0 ? (
   398→   398→              <div className="text-center py-12 text-slate-400 dark:text-slate-500 font-semibold text-xs">
   399→   399→                Chưa có hội thoại nào
   400→   400→              </div>
   401→   401→            ) : (
   402→   402→              chats.map((chat) => {
   403→   403→                const partner = getPartnerProfile(chat)
   404→   404→                const isActive = activeChat?.id === chat.id
   405→   405→                return (
   406→   406→                  <button
   407→   407→                    key={chat.id}
   408→   408→                    onClick={() => {
   409→   409→                      setActiveChat(chat)
   410→   410→                      navigate(`/chat/${chat.id}`)
   411→   411→                    }}
   412→   412→                    className={`w-full flex items-center gap-3 p-3.5 rounded-2xl transition-all text-left ${isActive
   413→   413→                        ? "bg-emerald-50 dark:bg-emerald-950/20 border border-emerald-100/50 dark:border-emerald-900/30"
   414→   414→                        : "hover:bg-slate-50 dark:hover:bg-slate-800/40 border border-transparent"
   415→   415→                      }`}
   416→   416→                  >
   417→   417→                    <Avatar className="h-10 w-10 border border-white dark:border-slate-800 shadow-sm shrink-0">
   418→   418→                      <AvatarImage src={partner.avatar_url} />
   419→   419→                      <AvatarFallback className="bg-emerald-100 text-emerald-700 font-bold">
   420→   420→                        {partner.full_name?.charAt(0).toUpperCase() || <User className="w-4 h-4" />}
   421→   421→                      </AvatarFallback>
   422→   422→                    </Avatar>
   423→   423→                    <div className="min-w-0 flex-1">
   424→   424→                      <h4 className={`text-sm font-bold truncate ${isActive ? "text-emerald-700 dark:text-emerald-400" : "text-slate-800 dark:text-slate-200"}`}>
   425→   425→                        {partner.full_name}
   426→   426→                      </h4>
   427→   427→                      <p className="text-[11px] text-slate-400 dark:text-slate-500 truncate font-semibold">
   428→   428→                        Sự kiện: {chat.events.title}
   429→   429→                      </p>
   430→   430→                    </div>
   431→   431→                  </button>
   432→   432→                )
   433→   433→              })
   434→   434→            )}
   435→   435→          </div>
   436→   436→        </div>
