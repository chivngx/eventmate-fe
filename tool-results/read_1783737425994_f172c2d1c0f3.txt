     1→import { useState, useEffect } from "react"
     2→import { useParams, useNavigate } from "react-router-dom"
     3→import { supabase } from "@/lib/supabase"
     4→import MainLayout from "@/components/layout/MainLayout"
     5→import { MapPin, Calendar, CheckCircle, XCircle, Clock3, Bookmark, Briefcase, Tag, DollarSign, Users, ExternalLink } from "lucide-react"
     6→import { Button } from "@/components/ui/button"

     9→
    10→export default function EventDetail() {
    11→    const { id } = useParams<{ id: string }>()
    12→    const navigate = useNavigate()

    15→    const [loading, setLoading] = useState(true)
    16→    const [role, setRole] = useState<string>("guest")
    17→    const [applyStatus, setApplyStatus] = useState<string | null>(null)
    18→    const [isApplying, setIsApplying] = useState(false)
    19→    const [isBookmarked, setIsBookmarked] = useState(false)
    20→
    21→    useEffect(() => {

    23→            setLoading(true)

    25→            let eventQuery = supabase
    26→                .from("events")
    27→                .select("*, profiles(id, full_name, avatar_url, slug, scale, address), danang_wards(name)")
    28→
    29→            if (isUuid) {
    30→                eventQuery = eventQuery.eq("id", id)
    31→            } else {
    32→                eventQuery = eventQuery.eq("slug", id)
    33→            }
    34→
    35→            const { data: eventData, error } = await eventQuery.maybeSingle()
    36→
    37→            if (eventData) {
    38→                setEvent(eventData)
    39→
    40→                // Tự động chuyển hướng URL từ ID dạng UUID sang dạng Slug SEO thân thiện
    41→                if (isUuid && eventData.slug) {
    42→                    navigate(`/jobs/${eventData.slug}`, { replace: true })
    43→                }
    44→
    45→                const { data: { user } } = await supabase.auth.getUser()
    46→
    47→                if (user) {
    48→                    const { data: profile } = await supabase.from("profiles").select("role").eq("id", user.id).maybeSingle()
    49→                    if (profile) setRole(profile.role)
    50→
    51→                    if (profile?.role === "student") {
    52→                        const { data: appData } = await supabase
    53→                            .from("applications")
    54→                            .select("status")
    55→                            .eq("event_id", eventData.id)
    56→                            .eq("student_id", user.id)
    57→                            .maybeSingle()
    58→
    59→                        if (appData) setApplyStatus(appData.status)
    60→
    61→                        const { data: bookmarkData } = await supabase
    62→                            .from("event_bookmarks")
    63→                            .select("id")
    64→                            .eq("event_id", eventData.id)
    65→                            .eq("student_id", user.id)
    66→                            .maybeSingle()
    67→
    68→                        if (bookmarkData) setIsBookmarked(true)
    69→                    }
    70→                }
    71→            } else {
    72→                console.error("Lỗi hoặc không tìm thấy sự kiện", error)
    73→            }
    74→            setLoading(false)
    75→        }
    76→
    77→        fetchEventDetails()
    78→    }, [id])
    79→
    80→    const handleApply = async () => {
    81→        setIsApplying(true)
    82→        const { data: { user } } = await supabase.auth.getUser()
    83→
    84→        if (!user) {
    85→            window.dispatchEvent(new CustomEvent("open-auth-modal", { detail: { mode: "login" } }))
    86→            return
    87→        }
    88→
    89→        const { error } = await supabase.from("applications").insert([
    90→            { event_id: event.id, student_id: user.id }
    91→        ])
    92→

    94→            showToast({ title: "Đã xảy ra lỗi", message: error.message, type: "error" })

    97→            showToast({ title: "Ứng tuyển thành công", message: "Đơn ứng tuyển của bạn đã được gửi. Vui lòng chờ BTC phê duyệt.", type: "success" })
    98→        }
    99→        setIsApplying(false)
   100→    }
   101→

   105→            window.dispatchEvent(new CustomEvent("open-auth-modal", { detail: { mode: "login" } }))
   106→            return
   107→        }
   108→
   109→        if (isBookmarked) {
   110→            const { error } = await supabase
   111→                .from("event_bookmarks")
   112→                .delete()
   113→                .eq("student_id", user.id)
   114→                .eq("event_id", event.id)
   115→
   116→            if (!error) {
   117→                setIsBookmarked(false)
   118→                showToast({ title: "Đã hủy lưu", message: "Đã hủy lưu tin tuyển dụng thành công.", type: "info" })
   119→            }

   121→        } else {
   122→            const { error } = await supabase
   123→                .from("event_bookmarks")
   124→                .insert([{ student_id: user.id, event_id: event.id }])
   125→
   126→            if (!error) {
   127→                setIsBookmarked(true)
   128→                showToast({ title: "Đã lưu tin", message: "Đã lưu tin tuyển dụng thành công.", type: "success" })
   129→            }
   130→            else showToast({ title: "Đã xảy ra lỗi", message: error.message, type: "error" })
   131→        }
   132→    }

   134→    if (loading) return <SkeletonEventDetail />
   135→

   137→        <MainLayout role="guest">

   139→                <h2 className="text-2xl font-black text-slate-900">Không tìm thấy sự kiện</h2>
   140→                <p className="text-slate-500 mt-2">Sự kiện này có thể đã bị xóa hoặc không tồn tại.</p>
   141→                <Button onClick={() => navigate(-1)} className="mt-4 rounded-xl bg-slate-900 text-white">Quay lại</Button>
   142→            </div>
   143→        </MainLayout>
   144→    )
   145→
   146→    const isPastDeadline = event.application_deadline ? new Date() > new Date(event.application_deadline) : false;
   147→    const disabledApply = isApplying || event.status !== 'upcoming';
   148→

   150→        if (role !== "student") return null

   155→                    <CheckCircle className="w-4 h-4 mr-2" /> Trúng tuyển
   156→                </button>
   157→            )
   158→        }
   159→        if (applyStatus === 'rejected') {
   160→            return (
   161→                <button disabled className="job-detail_info--actions-button button-primary btn-apply-job flex items-center justify-center rounded-md font-semibold bg-rose-50 border border-rose-100 text-rose-500 h-[40px] px-6 min-[1440px]:w-[569px] min-[1440px]:h-[40px] min-[1440px]:rounded-[6px] min-[1440px]:text-[14px] min-[1440px]:gap-[6px] min-[1440px]:leading-[22px] min-[1440px]:p-[8px_16px_8px_12px]">
   162→                    <XCircle className="w-4 h-4 mr-2" /> Chưa phù hợp
   163→                </button>
   164→            )

   170→                </button>

   172→        }
   173→
   174→        return (

   176→                onClick={handleApply}
   177→                disabled={disabledApply || isPastDeadline}
   178→                className="job-detail_info--actions-button button-primary open-apply-modal btn-apply-job flex items-center justify-center bg-[#00b14f] hover:bg-[#009a44] text-white font-semibold font-sans rounded-md transition-all active:scale-95 disabled:opacity-50 h-[40px] px-6 text-sm flex-1 cursor-pointer min-[1440px]:w-[569px] min-[1440px]:h-[40px] min-[1440px]:rounded-[6px] min-[1440px]:text-[14px] min-[1440px]:font-semibold min-[1440px]:gap-[6px] min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.175px] min-[1440px]:p-[8px_16px_8px_12px]"
   179→            >
   180→                {isApplying ? "Đang xử lý..." : isPastDeadline ? "Đã hết hạn nộp đơn" : event.status !== 'upcoming' ? "Đã đóng đăng ký" : "Ứng tuyển ngay"}
   181→            </button>
   182→        )
   183→    }
   184→
   185→    return (
   186→        <MainLayout role={role}>
   187→            <div className="max-w-6xl mx-auto pt-1 pb-6 px-4 animate-in fade-in slide-in-from-bottom-4 duration-500">
   188→                {/* Two Column Layout */}
   189→                <div className="job-detail_body flex flex-col lg:flex-row gap-6 items-start min-[1440px]:w-[1140px] min-[1440px]:gap-[28px] min-[1440px]:flex-row">
   190→                    {/* Left Column (Width: approx 760px on large screen) */}
   191→                    <div className="job-detail_body-left flex-1 w-full lg:max-w-[760px] space-y-6 min-[1440px]:w-[761px] min-[1440px]:gap-[28px] min-[1440px]:flex-col">
   192→                        {/* Box 1: Header / General Summary */}
   193→                        <div id="header-job-info" className="job-detail_box job-detail_info bg-white rounded-lg border border-slate-200 p-6 shadow-sm min-[1440px]:p-[20px_24px] min-[1440px]:rounded-[8px] min-[1440px]:gap-[16px] min-[1440px]:flex-col">
   194→                            <h1 className="job-detail_info--title text-xl md:text-2xl font-bold text-[#263a4d] leading-tight mb-4 min-[1440px]:mb-[16px] min-[1440px]:w-[713px] min-[1440px]:text-[20px] min-[1440px]:font-bold min-[1440px]:font-sans min-[1440px]:tracking-[-0.2px] min-[1440px]:leading-[28px]">
   195→                                {event.title}
   196→                            </h1>
   197→
   198→                            {/* Quick Info Grid */}
   199→                            <div className="job-detail_info--sections grid grid-cols-1 md:grid-cols-3 gap-4 border-t border-slate-100 pt-5 pb-6 min-[1440px]:w-[713px] min-[1440px]:h-[46px] min-[1440px]:flex-row min-[1440px]:items-center min-[1440px]:justify-between min-[1440px]:pt-0 min-[1440px]:pb-0 min-[1440px]:border-t-0 min-[1440px]:flex">
   200→                                <div className="job-detail_info--section section-salary flex items-center gap-3 min-[1440px]:h-[46px] min-[1440px]:flex-row min-[1440px]:items-center min-[1440px]:gap-[16px]">
   201→                                    <div className="job-detail_info--section-icon w-9 h-9 rounded-full bg-[#00b14f]/5 flex items-center justify-center shrink-0 min-[1440px]:w-[40px] min-[1440px]:h-[40px] min-[1440px]:rounded-[30px] min-[1440px]:p-[10px] min-[1440px]:flex min-[1440px]:flex-col min-[1440px]:justify-center min-[1440px]:items-center min-[1440px]:gap-[10px] min-[1440px]:bg-gradient-to-br min-[1440px]:from-[#00bf5d] min-[1440px]:to-[#00907c]">
   202→                                        <DollarSign className="w-4 h-4 text-[#00b14f] min-[1440px]:text-white" />
   203→                                    </div>
   204→                                    <div>
   205→                                        <p className="job-detail_info--section-content-title text-[12px] text-slate-400 font-medium min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#263a4d] min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.14px]">Quyền lợi / Lương</p>
   206→                                        <p className="job-detail_info--section-content-value text-[13px] text-[#212f3f] font-semibold truncate max-w-[180px] min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#212f3f] min-[1440px]:font-semibold min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.175px]" title={event.benefits || "Thỏa thuận"}>
   207→                                            {event.benefits || "Thỏa thuận"}
   208→                                        </p>
   209→                                    </div>
   210→                                </div>
   211→                                <div className="job-detail_info--section section-location flex items-center gap-3 min-[1440px]:flex-row min-[1440px]:items-center min-[1440px]:gap-[16px] min-[1440px]:flex-basis-0">
   212→                                    <div className="job-detail_info--section-icon w-9 h-9 rounded-full bg-[#00b14f]/5 flex items-center justify-center shrink-0 min-[1440px]:w-[40px] min-[1440px]:h-[40px] min-[1440px]:rounded-[30px] min-[1440px]:p-[10px] min-[1440px]:flex min-[1440px]:flex-col min-[1440px]:justify-center min-[1440px]:items-center min-[1440px]:gap-[10px] min-[1440px]:bg-gradient-to-br min-[1440px]:from-[#00bf5d] min-[1440px]:to-[#00907c]">
   213→                                        <MapPin className="w-4 h-4 text-[#00b14f] min-[1440px]:text-white" />
   214→                                    </div>
   215→                                    <div>
   216→                                        <p className="job-detail_info--section-content-title text-[12px] text-slate-400 font-medium min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#263a4d] min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.14px]">Khu vực</p>
   217→                                        <p className="job-detail_info--section-content-value text-[13px] text-[#212f3f] font-semibold truncate min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#212f3f] min-[1440px]:font-semibold min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.175px]" title={event.danang_wards?.name ? `P. ${event.danang_wards.name}` : "Đà Nẵng"}>
   218→                                            {event.danang_wards?.name ? `${event.danang_wards.name}` : "Đà Nẵng"}

   220→                                    </div>
   221→                                </div>
   222→                                <div className="job-detail_info--section section-eventdate flex items-center gap-3 min-[1440px]:flex-row min-[1440px]:items-center min-[1440px]:gap-[16px]">
   223→                                    <div className="job-detail_info--section-icon w-9 h-9 rounded-full bg-[#00b14f]/5 flex items-center justify-center shrink-0 min-[1440px]:w-[40px] min-[1440px]:h-[40px] min-[1440px]:rounded-[30px] min-[1440px]:p-[10px] min-[1440px]:flex min-[1440px]:flex-col min-[1440px]:justify-center min-[1440px]:items-center min-[1440px]:gap-[10px] min-[1440px]:bg-gradient-to-br min-[1440px]:from-[#00bf5d] min-[1440px]:to-[#00907c]">
   224→                                        <Calendar className="w-4 h-4 text-[#00b14f] min-[1440px]:text-white" />
   225→                                    </div>
   226→                                    <div>
   227→                                        <p className="job-detail_info--section-content-title text-[12px] text-slate-400 font-medium min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#263a4d] min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.14px]">Ngày diễn ra</p>
   228→                                        <p className="job-detail_info--section-content-value text-[13px] text-[#212f3f] font-semibold min-[1440px]:h-[22px] min-[1440px]:text-[14px] min-[1440px]:text-[#212f3f] min-[1440px]:font-semibold min-[1440px]:leading-[22px] min-[1440px]:tracking-[0.175px]">
